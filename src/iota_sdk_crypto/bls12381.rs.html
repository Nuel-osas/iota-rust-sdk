12const DST_G1: &[u8] = b"BLS_SIG_BLS12381G1_XMD:SHA-256_SSWU_RO_NUL_";

14#[derive(Debug)]
15#[allow(unused)]
16pub(crate) struct BlstError(pub(crate) blst::BLST_ERROR);

18impl std::fmt::Display for BlstError {

20        write!(f, "{self:?}")
21    }
22}

26#[derive(Clone)]
27pub struct Bls12381PrivateKey(SecretKey);

29impl std::fmt::Debug for Bls12381PrivateKey {

37#[cfg(test)]
38impl proptest::arbitrary::Arbitrary for Bls12381PrivateKey {
39    type Parameters = ();
40    type Strategy = proptest::strategy::BoxedStrategy<Self>;
41    fn arbitrary_with(_: Self::Parameters) -> Self::Strategy {
42        use proptest::strategy::Strategy;
43
44        proptest::arbitrary::any::<[u8; Self::LENGTH]>()
45            .prop_map(|bytes| {
46                let secret_key = SecretKey::key_gen(&bytes, &[]).unwrap();
47                Self(secret_key)
48            })
49            .boxed()
50    }
51}
52
53impl Bls12381PrivateKey {
54    /// The length of an bls12381 private key in bytes.
55    pub const LENGTH: usize = 32;
56
57    pub fn new(bytes: [u8; Self::LENGTH]) -> Result<Self, SignatureError> {
58        SecretKey::from_bytes(&bytes)
59            .map_err(BlstError)
60            .map_err(SignatureError::from_source)
61            .map(Self)
62    }
63
64    pub fn scheme(&self) -> SignatureScheme {
65        SignatureScheme::Bls12381
66    }
67
68    pub fn verifying_key(&self) -> Bls12381VerifyingKey {
69        let verifying_key = self.0.sk_to_pk();
70        Bls12381VerifyingKey(verifying_key)
71    }
72
73    pub fn public_key(&self) -> Bls12381PublicKey {
74        self.verifying_key().public_key()
75    }
76
77    pub fn generate<R>(mut rng: R) -> Self
78    where
79        R: rand_core::RngCore + rand_core::CryptoRng,
80    {
81        let mut buf: [u8; Self::LENGTH] = [0; Self::LENGTH];
82        rng.fill_bytes(&mut buf);
83        let secret_key = SecretKey::key_gen(&buf, &[]).unwrap();
84        Self(secret_key)
85    }
86
87    pub fn sign_checkpoint_summary(&self, summary: &CheckpointSummary) -> ValidatorSignature {
88        let message = summary.signing_message();
89        let signature = self.sign(&message);
90        ValidatorSignature {
91            epoch: summary.epoch,
92            public_key: self.public_key(),
93            signature,
94        }
95    }
96}
97
98impl Signer<Bls12381Signature> for Bls12381PrivateKey {
99    fn try_sign(&self, msg: &[u8]) -> Result<Bls12381Signature, SignatureError> {
100        let signature = self.0.sign(msg, DST_G1, &[]);
101        Ok(Bls12381Signature::new(signature.to_bytes()))
102    }
103}
104
105#[derive(Debug)]
106pub struct Bls12381VerifyingKey(pub(crate) PublicKey);
107
108impl Bls12381VerifyingKey {
109    pub fn new(public_key: &Bls12381PublicKey) -> Result<Self, SignatureError> {
110        PublicKey::key_validate(public_key.inner())
111            .map(Self)
112            .map_err(BlstError)
113            .map_err(SignatureError::from_source)
114    }
115
116    pub fn public_key(&self) -> Bls12381PublicKey {
117        Bls12381PublicKey::new(self.0.to_bytes())
118    }
119}
120
121impl Verifier<Bls12381Signature> for Bls12381VerifyingKey {
122    fn verify(&self, message: &[u8], signature: &Bls12381Signature) -> Result<(), SignatureError> {
123        let signature = Signature::sig_validate(signature.inner(), true)
124            .map_err(BlstError)
125            .map_err(SignatureError::from_source)?;
126
127        let err = signature.verify(true, message, DST_G1, &[], &self.0, false);
128        if err == blst::BLST_ERROR::BLST_SUCCESS {
129            Ok(())
130        } else {
131            Err(SignatureError::from_source(BlstError(err)))
132        }
133    }
134}
