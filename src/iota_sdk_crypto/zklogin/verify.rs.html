23const fn str_to_bn254(s: &str) -> Bn254FieldElement {
24    match Bn254FieldElement::from_str_radix_10(s) {
25        Ok(e) => e,
26        Err(_) => panic!("unable to convert bn254"),
27    }
28}

30const fn build_circom_g1([e0, e1, e2]: [&str; 3]) -> CircomG1 {
31    CircomG1([str_to_bn254(e0), str_to_bn254(e1), str_to_bn254(e2)])
32}
33
34const fn build_circom_g2([[e00, e01], [e10, e11], [e20, e21]]: [[&str; 2]; 3]) -> CircomG2 {
35    CircomG2([
36        [str_to_bn254(e00), str_to_bn254(e01)],
37        [str_to_bn254(e10), str_to_bn254(e11)],
38        [str_to_bn254(e20), str_to_bn254(e21)],
39    ])
40}

42fn circom_to_arkworks_g1(g1: &CircomG1) -> Result<G1Affine, SignatureError> {
43    let CircomG1([f0, f1, f2]) = g1;
44
45    let g1: G1Affine =
46        G1Projective::new_unchecked(bn254_to_fq(f0), bn254_to_fq(f1), bn254_to_fq(f2)).into();
47
48    if !g1.is_on_curve() || !g1.is_in_correct_subgroup_assuming_on_curve() {
49        return Err(SignatureError::from_source("invalid G1 input"));
50    }
51
52    Ok(g1)
53}
54
55fn circom_to_arkworks_g2(g2: &CircomG2) -> Result<G2Affine, SignatureError> {
56    let CircomG2([[f00, f01], [f10, f11], [f20, f21]]) = g2;
57
58    let g2: G2Affine = G2Projective::new_unchecked(
59        Fq2::new(bn254_to_fq(f00), bn254_to_fq(f01)),
60        Fq2::new(bn254_to_fq(f10), bn254_to_fq(f11)),
61        Fq2::new(bn254_to_fq(f20), bn254_to_fq(f21)),
62    )
63    .into();
64
65    if !g2.is_on_curve() || !g2.is_in_correct_subgroup_assuming_on_curve() {
66        return Err(SignatureError::from_source("invalid G2 input"));
67    }
68
69    Ok(g2)
70}

72fn bn254_to_fq(f: &Bn254FieldElement) -> Fq {
73    Fq::from_be_bytes_mod_order(f.padded())

76fn bn254_to_fr(f: &Bn254FieldElement) -> Fr {
77    Fr::from_be_bytes_mod_order(f.padded())

239    pub fn verify_zklogin(
240        &self,
241        jwk: &Jwk,
242        inputs: &ZkLoginInputs,
243        signature: &SimpleSignature,
244        max_epoch: u64,
245    ) -> Result<(), SignatureError> {
246        use base64ct::{Base64UrlUnpadded, Encoding};
247        // Decode modulus to bytes.
248        let modulus = Base64UrlUnpadded::decode_vec(&jwk.n)
249            .map_err(|e| SignatureError::from_source(e.to_string()))?;
250
251        let proof = zklogin_proof_to_arkworks(inputs.proof_points()).unwrap();
252        let input_hash = calculate_all_inputs_hash(inputs, signature, &modulus, max_epoch).unwrap();
253
254        self.verify_proof(&proof, &[input_hash])
255    }
256
257    fn verify_proof(
258        &self,
259        proof: &Proof<ark_bn254::Bn254>,
260        public_inputs: &[ark_bn254::Fr],
261    ) -> Result<(), SignatureError> {
262        use ark_snark::SNARK;
263
264        if ark_groth16::Groth16::<ark_bn254::Bn254>::verify_with_processed_vk(
265            &self.inner,
266            public_inputs,
267            proof,
268        )
269        .map_err(|e| SignatureError::from_source(e.to_string()))?
270        {
271            Ok(())
272        } else {
273            Err(SignatureError::from_source("Groth16 proof verify failed"))
274        }
275    }
276}

288/// Given a SimpleSignature convert the corresponding public key, prefixed with
289/// the signature scheme flag, to two Bn254Frs
290fn public_key_to_frs(signature: &SimpleSignature) -> Result<(Fr, Fr), SignatureError> {
291    // buf length of the longest public key secp256r1/secp256k1 of 33 bytes plus 1
292    // byte for the scheme
293    let mut buf = [0u8; 34];
294
295    buf[0] = signature.scheme().to_u8();
296
297    let buf = match signature {
298        SimpleSignature::Ed25519 { public_key, .. } => {
299            buf[1..Ed25519PublicKey::LENGTH + 1].copy_from_slice(public_key.inner());
300            &buf[..Ed25519PublicKey::LENGTH + 1]
301        }
302        SimpleSignature::Secp256k1 { public_key, .. } => {
303            buf[1..Secp256k1PublicKey::LENGTH + 1].copy_from_slice(public_key.inner());
304            &buf[..Secp256k1PublicKey::LENGTH + 1]
305        }
306        SimpleSignature::Secp256r1 { public_key, .. } => {
307            buf[1..Secp256r1PublicKey::LENGTH + 1].copy_from_slice(public_key.inner());
308            &buf[..Secp256r1PublicKey::LENGTH + 1]
309        }
310        _ => return Err(SignatureError::from_source("unknown signature scheme")),
311    };
312
313    // TODO this comment is wrong...
314    // Split the bytes deterministically such that the first element contains the
315    // first 128 bits of the hash, and the second element contains the latter
316    // ones.
317    let (first_half, second_half) = buf.split_at(buf.len() - 16);
318
319    let eph_public_key_0 = Fr::from_be_bytes_mod_order(first_half);
320    let eph_public_key_1 = Fr::from_be_bytes_mod_order(second_half);
321
322    Ok((eph_public_key_0, eph_public_key_1))
323}

325pub(crate) type U256 = bnum::BUintD8<32>;
326pub(crate) type U2048 = bnum::BUintD8<256>;

328const MAX_HEADER_LEN: u8 = 248;
329const PACK_WIDTH: u8 = 248;
330const MAX_EXT_ISS_LEN: u8 = 165;
331const MAX_ISS_LEN_B64: u8 = 4 * (1 + MAX_EXT_ISS_LEN / 3);
332
333/// Pads a stream of bytes and maps it to a field element
334pub fn hash_ascii_str_to_field(s: &str, max_size: u8) -> Result<Fr, SignatureError> {
335    let str_padded = str_to_padded_char_codes(s, max_size)?;
336    hash_to_field(&str_padded, 8, PACK_WIDTH)
337}
338
339fn str_to_padded_char_codes(s: &str, max_len: u8) -> Result<Vec<U256>, SignatureError> {
340    let arr: Vec<U256> = s.bytes().map(U256::from).collect();
341    pad_with_zeroes(arr, max_len)
342}
343
344fn pad_with_zeroes(in_arr: Vec<U256>, out_count: u8) -> Result<Vec<U256>, SignatureError> {
345    if in_arr.len() > out_count as usize {
346        return Err(SignatureError::from_source("in_arr too long"));
347    }
348    let mut padded = in_arr;
349    padded.resize(out_count as usize, U256::ZERO);
350    Ok(padded)
351}

353/// Maps a stream of bigints to a single field element. First we convert the
354/// base from inWidth to packWidth. Then we compute the poseidon hash of the
355/// "packed" input. input is the input vector containing equal-width big ints.
356/// inWidth is the width of each input element.
357fn hash_to_field<T: ToBits>(
358    input: &[T],
359    in_width: u16,
360    pack_width: u8,
361) -> Result<Fr, SignatureError> {
362    let packed = convert_base(input, in_width, pack_width)?;
363
364    POSEIDON.hash(&packed).map_err(SignatureError::from_source)
365}
366
367/// Helper function to pack field elements from big ints.
368fn convert_base<T: ToBits>(
369    in_arr: &[T],
370    in_width: u16,
371    out_width: u8,
372) -> Result<Vec<Fr>, SignatureError> {
373    if out_width == 0 {
374        return Err(SignatureError::from_source("invalid input"));
375    }
376    let bits = big_int_array_to_bits(in_arr, in_width as usize)?;
377    let mut packed: Vec<Fr> = bits
378        .rchunks(out_width as usize)
379        .map(|chunk| {
380            Fr::from_be_bytes_mod_order(U256::from_radix_be(chunk, 2).unwrap().to_be().digits())
381        })
382        .collect();
383    packed.reverse();
384    match packed.len() != (in_arr.len() * in_width as usize).div_ceil(out_width as usize) {
385        true => Err(SignatureError::from_source("invalid input")),
386        false => Ok(packed),
387    }
388}

390/// Convert a big int array to a bit array with 0 paddings.
391fn big_int_array_to_bits<T: ToBits>(
392    integers: &[T],
393    intended_size: usize,
394) -> Result<Vec<u8>, SignatureError> {
395    use std::cmp::Ordering::{Equal, Greater, Less};
396
397    use itertools::Itertools;
398
399    integers
400        .iter()
401        .map(|integer| {
402            let bits = integer.to_bits();
403            match bits.len().cmp(&intended_size) {
404                Less => {
405                    let extra_bits = intended_size - bits.len();
406                    let mut padded = vec![0; extra_bits];
407                    padded.extend(bits);
408                    Ok(padded)
409                }
410                Equal => Ok(bits),
411                Greater => Err(SignatureError::from_source("invalid input")),
412            }
413        })
414        .flatten_ok()
415        .collect()
416}

434/// Calculate the poseidon hash from selected fields from inputs, along with the
435/// ephemeral pubkey.
436pub fn calculate_all_inputs_hash(
437    inputs: &ZkLoginInputs,
438    signature: &SimpleSignature,
439    modulus: &[u8],
440    max_epoch: u64,
441) -> Result<Fr, SignatureError> {
442    if inputs.header_base64().len() > MAX_HEADER_LEN as usize {
443        return Err(SignatureError::from_source("header too long"));
444    }
445
446    let (first, second) = public_key_to_frs(signature)?;
447
448    let address_seed = bn254_to_fr(inputs.address_seed());
449    let max_epoch_f = Fr::from_be_bytes_mod_order(U256::from(max_epoch).to_be().digits());
450    let index_mod_4_f = Fr::from_be_bytes_mod_order(
451        U256::from(inputs.iss_base64_details().index_mod_4)
452            .to_be()
453            .digits(),
454    );
455
456    let iss_base64_f =
457        hash_ascii_str_to_field(&inputs.iss_base64_details().value, MAX_ISS_LEN_B64)?;
458    let header_f = hash_ascii_str_to_field(inputs.header_base64(), MAX_HEADER_LEN)?;
459    let modulus_f = hash_to_field(&[U2048::from_be_slice(modulus).unwrap()], 2048, PACK_WIDTH)?;
460
461    POSEIDON
462        .hash(&[
463            first,
464            second,
465            address_seed,
466            max_epoch_f,
467            iss_base64_f,
468            index_mod_4_f,
469            header_f,
470            modulus_f,
471        ])
472        .map_err(SignatureError::from_source)
473}

475/// Calculate the IOTA address based on address seed and address params.
476#[allow(unused)]
477fn gen_address_seed(
478    salt: &str,
479    name: &str,  // i.e. "sub"
480    value: &str, // i.e. the sub value
481    aud: &str,   // i.e. the client ID
482) -> Result<String, SignatureError> {
483    let salt_hash = POSEIDON
484        .hash(&[bn254_to_fr(
485            &Bn254FieldElement::from_str(salt).map_err(SignatureError::from_source)?,
486        )])
487        .map_err(SignatureError::from_source)?;
488    gen_address_seed_with_salt_hash(salt_hash, name, value, aud)
489}
490
491const MAX_KEY_CLAIM_NAME_LENGTH: u8 = 32;
492const MAX_KEY_CLAIM_VALUE_LENGTH: u8 = 115;
493const MAX_AUD_VALUE_LENGTH: u8 = 145;
494
495/// Same as [`gen_address_seed`] but takes the poseidon hash of the salt as
496/// input instead of the salt.
497pub(crate) fn gen_address_seed_with_salt_hash(
498    salt_hash: Fr,
499    name: &str,  // i.e. "sub"
500    value: &str, // i.e. the sub value
501    aud: &str,   // i.e. the client ID
502) -> Result<String, SignatureError> {
503    Ok(POSEIDON
504        .hash(&[
505            hash_ascii_str_to_field(name, MAX_KEY_CLAIM_NAME_LENGTH)?,
506            hash_ascii_str_to_field(value, MAX_KEY_CLAIM_VALUE_LENGTH)?,
507            hash_ascii_str_to_field(aud, MAX_AUD_VALUE_LENGTH)?,
508            salt_hash,
509        ])
510        .map_err(SignatureError::from_source)?
511        .to_string())
512}
