8use iota_types::{Identifier, ObjectId, ObjectReference, TypeTag};

10/// An identifier indicating the unresolved index of an input.
11pub type InputId = usize;
12
13#[derive(Debug, Clone, PartialEq, Eq)]
14pub struct Input {
15    pub kind: InputKind,
16    pub is_gas: bool,
17}
18
19impl Input {
20    pub fn object_id(&self) -> Option<&ObjectId> {
21        match &self.kind {
22            InputKind::ImmutableOrOwned(object_id)
23            | InputKind::Shared { object_id, .. }
24            | InputKind::Receiving(object_id) => Some(object_id),
25            InputKind::Input(input) => match input {
26                iota_types::Input::Pure { .. } => None,
27                iota_types::Input::ImmutableOrOwned(ObjectReference { object_id, .. })
28                | iota_types::Input::Shared { object_id, .. }
29                | iota_types::Input::Receiving(ObjectReference { object_id, .. }) => {
30                    Some(object_id)
31                }
32                _ => unimplemented!("a new enum variant was added and needs to be handled"),
33            },
34        }
35    }
36}
37
38#[derive(Debug, Clone, PartialEq, Eq)]
39#[non_exhaustive]
40pub enum InputKind {
41    ImmutableOrOwned(ObjectId),
42    Shared { object_id: ObjectId, mutable: bool },
43    Receiving(ObjectId),
44    Input(iota_types::Input),
45}
46
47impl InputKind {
48    pub fn object_id(&self) -> Option<ObjectId> {
49        if let Self::ImmutableOrOwned(object_id)
50        | Self::Receiving(object_id)
51        | Self::Shared { object_id, .. }
52        | Self::Input(
53            iota_types::Input::ImmutableOrOwned(ObjectReference { object_id, .. })
54            | iota_types::Input::Receiving(ObjectReference { object_id, .. })
55            | iota_types::Input::Shared { object_id, .. },
56        ) = self
57        {
58            Some(*object_id)
59        } else {
60            None
61        }
62    }
63}

65#[derive(Debug, Clone, derive_more::From)]
66#[non_exhaustive]
67pub enum Command {
68    MoveCall(MoveCall),
69    TransferObjects(TransferObjects),
70    SplitCoins(SplitCoins),
71    MergeCoins(MergeCoins),
72    Publish(Publish),
73    MakeMoveVector(MakeMoveVector),
74    Upgrade(Upgrade),
75}
76
77impl Command {
78    pub fn resolve(self, input_map: &HashMap<InputId, u16>) -> iota_types::Command {
79        match self {
80            Command::MoveCall(move_call) => {
81                iota_types::Command::MoveCall(move_call.resolve(input_map))
82            }
83            Command::TransferObjects(transfer_objects) => {
84                iota_types::Command::TransferObjects(transfer_objects.resolve(input_map))
85            }
86            Command::SplitCoins(split_coins) => {
87                iota_types::Command::SplitCoins(split_coins.resolve(input_map))
88            }
89            Command::MergeCoins(merge_coins) => {
90                iota_types::Command::MergeCoins(merge_coins.resolve(input_map))
91            }
92            Command::Publish(publish) => iota_types::Command::Publish(publish.resolve()),
93            Command::MakeMoveVector(make_move_vector) => {
94                iota_types::Command::MakeMoveVector(make_move_vector.resolve(input_map))
95            }
96            Command::Upgrade(upgrade) => iota_types::Command::Upgrade(upgrade.resolve(input_map)),
97        }
98    }
99}

101#[derive(Debug, Clone)]
102pub struct MoveCall {
103    pub package: ObjectId,
104    pub module: Identifier,
105    pub function: Identifier,
106    pub type_arguments: Vec<TypeTag>,
107    pub arguments: Vec<Argument>,
108}
109
110impl MoveCall {
111    fn resolve(self, input_map: &HashMap<InputId, u16>) -> iota_types::MoveCall {
112        iota_types::MoveCall {
113            package: self.package,
114            module: self.module,
115            function: self.function,
116            type_arguments: self.type_arguments,
117            arguments: self
118                .arguments
119                .into_iter()
120                .map(|c| c.resolve(input_map))
121                .collect(),
122        }
123    }
124}

236#[derive(Debug, Clone, Copy)]
237#[non_exhaustive]
238pub enum Argument {
239    Gas,
240    Input(InputId),
241    Result(u16),
242    NestedResult(u16, u16),
243}
244
245impl Argument {
246    fn resolve(self, input_map: &HashMap<InputId, u16>) -> iota_types::Argument {
247        match self {
248            Argument::Gas => iota_types::Argument::Gas,
249            Argument::Input(i) => input_map
250                .get(&i)
251                .map(|i| iota_types::Argument::Input(*i))
252                .unwrap_or(iota_types::Argument::Gas),
253            Argument::Result(i) => iota_types::Argument::Result(i),
254            Argument::NestedResult(i1, i2) => iota_types::Argument::NestedResult(i1, i2),
255        }
256    }
