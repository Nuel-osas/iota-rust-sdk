4//! # IOTA Transaction Builder
5//!
6//! This crate contains the [TransactionBuilder], which allows for simple
7//! construction of Programmable Transactions which can be executed on the IOTA
8//! network.
9//!
10//! The builder is designed to allow for a lot of flexibility while also
11//! reducing the necessary boilerplate code. It uses a type-state pattern to
12//! ensure the proper flow through the various functions. It is chainable via
13//! mutable references.
14//!
15//! ## Online vs. Offline Builder
16//!
17//! The Transaction Builder can be used with or without a
18//! [GraphQLClient](iota_graphql_client::Client). When one is provided via the
19//! [with_client](TransactionBuilder::with_client) method, the resulting builder
20//! will use it to find and validate provided IDs.
21//!
22//! ### Example with Client Resolution
23//!
24//! ```
25//! # use std::str::FromStr;
26//! use iota_graphql_client::Client;
27//! use iota_sdk_transaction_builder::TransactionBuilder;
28//! use iota_types::{Address, ObjectId, Transaction};
29//!
30//! # #[tokio::main(flavor = "current_thread")]
31//! # async fn main() -> eyre::Result<()> {
32//!
33//! let sender =
34//!     Address::from_str("0x611830d3641a68f94a690dcc25d1f4b0dac948325ac18f6dd32564371735f32c")?;
35//! let to_address =
36//!     Address::from_str("0x0000a4984bd495d4346fa208ddff4f5d5e5ad48c21dec631ddebc99809f16900")?;
37//!
38//! let mut builder = TransactionBuilder::new(sender).with_client(Client::new_devnet());
39//!
40//! let coin =
41//!     ObjectId::from_str("0x8ef4259fa2a3499826fa4b8aebeb1d8e478cf5397d05361c96438940b43d28c9")?;
42//!
43//! builder.send_coins([coin], to_address, 50000000000u64);
44//!
45//! let txn: Transaction = builder.finish().await?;
46//! # Ok(())
47//! # }
48//! ```
49//!
50//! ### Example without Client Resolution
51//!
52//! ```
53//! # use std::str::FromStr;
54//! use iota_sdk_transaction_builder::TransactionBuilder;
55//! use iota_types::{Address, Digest, ObjectId, ObjectReference, Transaction};
56//!
57//! let sender =
58//!     Address::from_str("0x611830d3641a68f94a690dcc25d1f4b0dac948325ac18f6dd32564371735f32c")?;
59//! let to_address =
60//!     Address::from_str("0x0000a4984bd495d4346fa208ddff4f5d5e5ad48c21dec631ddebc99809f16900")?;
61//!
62//! let mut builder = TransactionBuilder::new(sender);
63//!
64//! let coin = ObjectReference {
65//!     object_id: ObjectId::from_str(
66//!         "0x8ef4259fa2a3499826fa4b8aebeb1d8e478cf5397d05361c96438940b43d28c9",
67//!     )?,
68//!     digest: Digest::from_str("4jJMQScR4z5kK3vchvDEFYTiCkZPEYdvttpi3iTj1gEW")?,
69//!     version: 435090179,
70//! };
71//! let gas_coin = ObjectReference {
72//!     object_id: ObjectId::from_str(
73//!         "0xd04077fe3b6fad13b3d4ed0d535b7ca92afcac8f0f2a0e0925fb9f4f0b30c699",
74//!     )?,
75//!     digest: Digest::from_str("8ahH5RXFnK1jttQEWTypYX7MRzLuQDEXk7fhMHCyZekX")?,
76//!     version: 473053810,
77//! };
78//!
79//! builder
80//!     .send_coins([coin], to_address, 50000000000u64)
81//!     .gas([gas_coin])
82//!     .gas_budget(1000000000)
83//!     .gas_price(100);
84//!
85//! let txn: Transaction = builder.finish()?;
86//! # Result::<_, eyre::Error>::Ok(())
87//! ```
88//!
89//! NOTE: It is possible to provide an [ObjectId](iota_types::ObjectId) to an
90//! offline client builder, but this will cause the builder to fail when calling
91//! `finish`.
92//!
93//! ## Methods
94//!
95//! There are three kinds of methods available:
96//!
97//! ### Commands
98//!
99//! Each command method adds one or more commands to the final transaction. Some
100//! commands have optional follow-up methods. All command results can be
101//! assigned a name via [assign](TransactionBuilder::assign). Assigning a name
102//! to a command allows them to be used later in the transaction via the
103//! [assigned] method.
104//!
105//! - [move_call](TransactionBuilder::move_call): Call a move function.
106//!     - `arguments`: Add arguments to the move call.
107//!     - `generics`: Add generic types to the move call using types that
108//!       implement [MoveType](types::MoveType).
109//!     - `type_tags`: Add generic types directly using the
110//!       [TypeTag](iota_types::TypeTag).
111//! - [send_iota](TransactionBuilder::send_iota): Send IOTA coins to a recipient
112//!   address.
113//! - [send_coins](TransactionBuilder::send_coins): Send coins of any type to a
114//!   recipient address.
115//! - [merge_coins](TransactionBuilder::merge_coins): Merge a list of coins into
116//!   a single primary coin.
117//! - [split_coins](TransactionBuilder::split_coins): Split a coin into coins of
118//!   various amounts.
119//! - [transfer_objects](TransactionBuilder::transfer_objects): Send objects to
120//!   a recipient address.
121//! - [publish](TransactionBuilder::publish): Publish a move package.
122//!     - `package_id`: Name the package ID returned by the publish call.
123//! - [upgrade](TransactionBuilder::upgrade): Upgrade a move package.
124//! - [make_move_vec](TransactionBuilder::make_move_vec): Create a move
125//!   `vector`.
126//!
127//! ### Metadata
128//!
129//! These methods set various metadata which may be needed for the execution.
130//!
131//! - [gas](TransactionBuilder::gas): Add gas coins to pay for the execution.
132//! - [gas_budget](TransactionBuilder::gas_budget): Set the maximum gas budget
133//!   to spend.
134//! - [gas_price](TransactionBuilder::gas_price): Set the gas price.
135//! - [sponsor](TransactionBuilder::sponsor): Set the gas sponsor address.
136//! - [gas_station_sponsor](TransactionBuilder::gas_station_sponsor): Set the
137//!   gas station URL. See [Gas Station](crate#gas-station) for more info.
138//! - [expiration](TransactionBuilder::expiration): Set the transaction
139//!   expiration epoch.
140//!
141//! ### Other
142//!
143//! Many other methods exist, either to get data or allow for development on top
144//! of the builder. Typically, these methods should not be needed, but they are
145//! made available for special circumstances.
146//!
147//! - [apply_argument](TransactionBuilder::apply_argument)
148//! - [apply_arguments](TransactionBuilder::apply_arguments)
149//! - [input](TransactionBuilder::input)
150//! - [pure_bytes](TransactionBuilder::pure_bytes)
151//! - [pure](TransactionBuilder::pure)
152//! - [command](TransactionBuilder::command)
153//! - [assigned_command](TransactionBuilder::assigned_command)
154//!
155//! ## Finalization and Execution
156//!
157//! There are several ways to finish the builder. First, the
158//! [finish](TransactionBuilder::finish) method can be used to return the
159//! resulting [Transaction](iota_types::Transaction), which can be manually
160//! serialized, executed, etc.
161//!
162//! Additionally, when a client is provided, the builder can directly
163//! [dry_run](TransactionBuilder::dry_run) or
164//! [execute](TransactionBuilder::execute) the transaction.
165//!
166//! When the transaction is resolved, the builder will try to ensure a valid
167//! state by de-duplicating and converting appropriate inputs into references to
168//! the gas coin. This means that the same input can be passed multiple times
169//! and the final transaction will only contain one instance. However, in some
170//! cases an invalid state can still be reached. For instance, if a coin is used
171//! both for gas and as part of a group of coins, i.e. when transferring
172//! objects, the transaction can not possibly be valid.
173//!
174//! ### Defaults
175//!
176//! When a client is provided, the builder can set some values by default. The
177//! following are the default behaviors for each metadata value.
178//!
179//! - Gas: One page of coins owned by the sender.
180//! - Gas Budget: A dry run will be used to estimate.
181//! - Gas Price: The current reference gas price.
182//!
183//! ## Gas Station
184//!
185//! The Transaction Builder supports executing via a
186//! [Gas Station](https://github.com/iotaledger/gas-station). To do so, the URL
187//! must be provided via
188//! [gas_station_sponsor](TransactionBuilder::gas_station_sponsor). Additional
189//! configuration can then be provided via
190//! [gas_reservation_duration](TransactionBuilder::gas_reservation_duration) and
191//! [add_gas_station_header](TransactionBuilder::add_gas_station_header).
192//!
193//! By default the request will contain the header `Content-Type:
194//! application/json`
195//!
196//! When this data has been set, calling [execute](TransactionBuilder::execute)
197//! will request gas from and send the resulting transaction to this endpoint
198//! instead of using the GraphQL client.
199//!
200//! ## Traits and Helpers
201//!
202//! This crate provides several traits which enable the functionality of the
203//! builder. Often, when providing arguments, functions will accept either a
204//! single [PTBArgument] or a [PTBArgumentList].
205//!
206//! [PTBArgument] is implemented for any type implementing
207//! [MoveArg](types::MoveArg) as well as:
208//! - [unresolved::Argument]: Arguments returned by various builder functions.
209//!   Distinct from [iota_types::Argument], which cannot be used.
210//! - [Input](iota_types::Input): A resolved input.
211//! - [ObjectId](iota_types::ObjectId): An object's ID. Can only be used when a
212//!   client is provided. This will be assumed immutable or owned.
213//! - [ObjectReference](iota_types::ObjectReference): An object's reference.
214//!   This will be assumed immutable or owned.
215//! - [Assigned](builder::ptb_arguments::Assigned): A reference to the result of
216//!   a previous assigned command, set with
217//!   [assign](TransactionBuilder::assign).
218//! - [Shared]: Allows specifying shared immutable move objects.
219//! - [SharedMut]: Allows specifying shared mutable move objects.
220//! - [Receiving]: Allows specifying receiving move objects.
221//!
222//! [PTBArgumentList] is implemented for collection types, and represents a set
223//! of arguments. For move calls, this enables tuples of rust values to
224//! represent the parameters defined in the smart contract. For calls like
225//! [merge_coins](TransactionBuilder::merge_coins), this can represent a list of
226//! coins.
227//!
228//! [MoveArg](types::MoveArg) represents types that can be serialized and
229//! provided to the transaction as pure bytes.
230//!
231//! [MoveType](types::MoveType) defines the type tag for a rust type, so that it
232//! can be used for generic arguments.
233//!
234//! ### Example
235//!
236//! The following function is defined in move in `vec_map`:
237//!
238//! ```ignore
239//! public fun from_keys_values<K: copy, V>(mut keys: vector<K>, mut values: vector<V>): VecMap<K, V>
240//! ```
241//!
242//! ```ignore
243//! builder
244//!     .move_call(Address::TWO, "vec_map", "from_keys_values")
245//!     .generics::<(Address, u64)>()
246//!     .arguments(([address1, address2], [10000000u64, 20000000u64]));
247//! ```
248//!
249//! ### Custom Type
250//!
251//! In order to use a custom type, implement [MoveType](types::MoveType) and
252//! [MoveArg](types::MoveArg).
253//!
254//! ```
255//! # use std::str::FromStr;
256//! # use iota_sdk_transaction_builder::types::{MoveArg, MoveType, PureBytes};
257//! # use iota_types::TypeTag;
258//! #[derive(serde::Serialize)]
259//! struct MyStruct {
260//!     val1: String,
261//!     val2: u64,
262//! }
263//!
264//! impl MoveType for MyStruct {
265//!     fn type_tag() -> TypeTag {
266//!         TypeTag::from_str("0x0::my_module::MyStruct").unwrap()
267//!     }
268//! }
269//!
270//! impl MoveArg for MyStruct {
271//!     fn pure_bytes(self) -> PureBytes {
272//!         PureBytes(bcs::to_bytes(&self).unwrap())
273//!     }
274//! }
275//! ```

315    /// This is used to read the json file that contains the modules/deps/digest
316    /// generated with iota move build --dump-bytecode-as-base64 on the
317    /// `test_example_v1 and test_example_v2` projects in the tests
318    /// directory. The json files are generated automatically when running
319    /// `make test-with-localnet` in the root of the
320    /// iota-sdk-transaction-builder crate.
321    fn move_package_data(file: &str) -> MovePackageData {
322        let data = std::fs::read_to_string(file)
323            .with_context(|| {
324                format!(
325                    "Failed to read {file}. \
326                    Run `make test-with-localnet` from the root of the repository that will \
327                    generate the right json files with the package data and then run the tests."
328                )
329            })
330            .unwrap();
331        serde_json::from_str(&data).unwrap()
332    }

334    /// Generate a random private key and its corresponding address
335    fn helper_address_pk() -> (Address, Ed25519PrivateKey) {
336        let pk = Ed25519PrivateKey::generate(rand::thread_rng());
337        let address = pk.public_key().derive_address();
338        (address, pk)

341    /// Helper to:
342    /// - generate a private key and its corresponding address
343    /// - set the sender for the tx to this newly created address
344    /// - set gas price
345    /// - set gas budget
346    /// - call faucet which returns 5 coin objects
347    /// - set the gas object (last coin from the list of the 5 objects returned
348    ///   by faucet)
349    /// - return the address, private key, and coins.
350    ///
351    /// NB! This assumes that these tests run on a network whose faucet returns
352    /// 5 coins per each faucet request.
353    async fn helper_setup() -> (
354        TransactionBuilder<Client>,
355        Address,
356        Ed25519PrivateKey,
357        Vec<CoinInfo>,
358    ) {
359        let (address, pk) = helper_address_pk();
360        let client = Client::new_localnet();
361        let mut tx = TransactionBuilder::new(address).with_client(client.clone());
362        let coins = FaucetClient::new_localnet()
363            .request_and_wait(address)
364            .await
365            .unwrap()
366            .unwrap()
367            .sent;
368        let tx_digest = coins.first().unwrap().transfer_tx_digest;
369        wait_for_tx(&client, tx_digest).await;
370
371        let gas = coins.last().unwrap().id;
372        tx.gas([gas]);
373
374        (tx, address, pk, coins)
375    }

377    /// Wait for the transaction to be finalized and indexed. This queries the
378    /// GraphQL server until it retrieves the requested transaction.
379    async fn wait_for_tx(client: &Client, digest: Digest) {
380        while !client.is_tx_finalized(digest).await.unwrap() {
381            tokio::time::sleep(tokio::time::Duration::from_millis(100)).await;
382        }
383    }

385    /// Wait for the transaction to be finalized and indexed, and check the
386    /// effects' to ensure the transaction was successfully executed.
387    async fn check_effects_status_success(effects: Result<TransactionEffects, Error>) {
388        assert!(effects.is_ok(), "Execution failed. Effects: {effects:?}");
389        // check that it succeeded
390        let status = effects.unwrap();
391        let expected_status = ExecutionStatus::Success;
392        assert_eq!(&expected_status, status.status());
393    }

553    #[tokio::test]
554    async fn test_upgrade() {
555        let (mut tx, address, pk, coins) = helper_setup().await;
556
557        let package = move_package_data("package_test_example_v2.json");
558        tx.publish(package)
559            .upgrade_cap("cap")
560            .transfer_objects(address, [assigned("cap")]);
561
562        let effects = tx.execute(&pk, WaitForTx::Finalized).await;
563        let mut package_id: Option<ObjectId> = None;
564        let mut created_objs = vec![];
565        if let Ok(ref effects) = effects {
566            match effects {
567                TransactionEffects::V1(e) => {
568                    for obj in e.changed_objects.clone() {
569                        if obj.id_operation == IdOperation::Created {
570                            let change = obj.output_state;
571                            match change {
572                                iota_types::ObjectOut::PackageWrite { .. } => {
573                                    package_id = Some(obj.object_id);
574                                }
575                                iota_types::ObjectOut::ObjectWrite { .. } => {
576                                    created_objs.push(obj.object_id);
577                                }
578                                _ => {}
579                            }
580                        }
581                    }
582                }
583                _ => unimplemented!("a new enum variant was added and needs to be handled"),
584            }
585        }
586        check_effects_status_success(effects).await;
587
588        let client = Client::new_localnet();
589        let mut tx = TransactionBuilder::new(address).with_client(&client);
590        let mut upgrade_cap = None;
591        for o in created_objs {
592            let obj = client.object(o, None).await.unwrap().unwrap();
593            match obj.object_type() {
594                ObjectType::Struct(x) if x.name() == "UpgradeCap" => {
595                    upgrade_cap = Some(obj.object_id());
596                    break;
597                }
598                _ => {}
599            };
600        }
601
602        let updated_package = move_package_data("package_test_example_v2.json");
603
604        // we need this ticket to authorize the upgrade
605        tx.move_call(Address::FRAMEWORK, "package", "authorize_upgrade")
606            .arguments((
607                upgrade_cap.unwrap(),
608                UpgradePolicy::Compatible as u8,
609                updated_package.digest,
610            ))
611            .assign("ticket");
612        // now we can upgrade the package
613        let receipt = tx
614            .upgrade(package_id.unwrap(), updated_package, assigned("ticket"))
615            .arg();
616
617        // commit the upgrade
618        tx.move_call(Address::FRAMEWORK, "package", "commit_upgrade")
619            .arguments((upgrade_cap.unwrap(), receipt));
620
621        tx.gas([coins.last().unwrap().id]);
622
