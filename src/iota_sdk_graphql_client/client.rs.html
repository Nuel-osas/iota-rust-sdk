10use crate::{
11    error::{Error, Result},

16pub(crate) const DEFAULT_ITEMS_PER_PAGE: i32 = 10;
17pub(crate) const MAINNET_HOST: &str = "https://graphql.mainnet.iota.cafe";
18pub(crate) const TESTNET_HOST: &str = "https://graphql.testnet.iota.cafe";
19pub(crate) const DEVNET_HOST: &str = "https://graphql.devnet.iota.cafe";
20pub(crate) const LOCAL_HOST: &str = "http://localhost:9125/graphql";
21pub(crate) static USER_AGENT: &str =
22    concat!(env!("CARGO_PKG_NAME"), "/", env!("CARGO_PKG_VERSION"));
23
24/// Helper function to convert a GraphQL response to a Result.
25pub(crate) fn response_to_err<T>(response: GraphQlResponse<T>) -> Result<T, Error> {
26    match (response.data, response.errors) {
27        (Some(data), None) => Ok(data),
28        (None, Some(errors)) => Err(Error::graphql_error(errors)),
29        _ => unreachable!(
30            "Either data or errors must be present in a GraphQL response, but not both"
31        ),
32    }
33}

35/// The GraphQL client for interacting with the IOTA blockchain.
36/// By default, it uses the `reqwest` crate as the HTTP client.
37#[derive(Debug, Clone)]
38pub struct Client {
39    /// The URL of the GraphQL server.
40    pub(crate) rpc: Url,
41    /// The reqwest client.
42    pub(crate) inner: reqwest::Client,
43    pub(crate) service_config: std::sync::OnceLock<ServiceConfig>,
44}
45
46impl Client {
47    /// Create a new GraphQL client with the provided server address.
48    pub fn new(server: &str) -> Result<Self> {
49        let rpc = reqwest::Url::parse(server)?;
50
51        let client = Client {
52            rpc,
53            inner: reqwest::Client::builder().user_agent(USER_AGENT).build()?,
54            service_config: Default::default(),
55        };
56        Ok(client)
57    }
58
59    /// Create a new GraphQL client connected to the `mainnet` GraphQL server:
60    /// {MAINNET_HOST}.
61    pub fn new_mainnet() -> Self {
62        Self::new(MAINNET_HOST).expect("Invalid mainnet URL")
63    }
64
65    /// Create a new GraphQL client connected to the `testnet` GraphQL server:
66    /// {TESTNET_HOST}.
67    pub fn new_testnet() -> Self {
68        Self::new(TESTNET_HOST).expect("Invalid testnet URL")
69    }
70
71    /// Create a new GraphQL client connected to the `devnet` GraphQL server:
72    /// {DEVNET_HOST}.
73    pub fn new_devnet() -> Self {
74        Self::new(DEVNET_HOST).expect("Invalid devnet URL")
75    }
76
77    /// Create a new GraphQL client connected to a `localnet` GraphQL server:
78    /// {LOCAL_HOST}.
79    pub fn new_localnet() -> Self {
80        Self::new(LOCAL_HOST).expect("Invalid localhost URL")
81    }
82
83    /// Return the URL for the GraphQL server.
84    pub(crate) fn rpc_server(&self) -> &Url {
85        &self.rpc
86    }
87
88    /// Set the server address for the GraphQL client. It should be a
89    /// valid URL with a host and optionally a port number.
90    pub fn set_rpc_server(&mut self, server: &str) -> Result<()> {
91        let rpc = reqwest::Url::parse(server)?;
92        self.rpc = rpc;
93        Ok(())
94    }
95
96    /// Get the GraphQL service configuration, including complexity limits, read
97    /// and mutation limits, supported versions, and others.
98    pub async fn service_config(&self) -> Result<&ServiceConfig> {
99        // If the value is already initialized, return it
100        if let Some(service_config) = self.service_config.get() {
101            return Ok(service_config);
102        }
103
104        // Otherwise, fetch and initialize it
105        let operation = ServiceConfigQuery::build(());
106        let response = self.run_query(&operation).await?;
107
108        let service_config = self
109            .service_config
110            .get_or_init(move || response.service_config);
111
112        Ok(service_config)
113    }
114
115    /// Run a query on the GraphQL server and return the response.
116    /// This method returns [`cynic::GraphQlResponse`]  over the query type `T`,
117    /// and it is intended to be used with custom queries.
118    pub async fn run_query<T, V>(&self, operation: &Operation<T, V>) -> Result<T>
119    where
120        T: serde::de::DeserializeOwned,
121        V: serde::Serialize,
122    {
123        response_to_err(
124            self.inner
125                .post(self.rpc_server().clone())
126                .json(&operation)
127                .send()
128                .await?
129                .json::<GraphQlResponse<T>>()
130                .await?,
131        )
132    }
133
134    /// Run a JSON query on the GraphQL server and return the response.
135    /// This method expects a JSON map holding the GraphQL query string and
136    /// matching GraphQL variables. It returns a [`cynic::GraphQlResponse`]
137    /// wrapping a [`serde_json::Value`]. In general, it is recommended to use
138    /// [`run_query`](`Self::run_query`) which guarantees valid GraphQL
139    /// query syntax and returns a proper response type.
140    pub async fn run_query_from_json(
141        &self,
142        json: serde_json::Map<String, serde_json::Value>,
143    ) -> Result<GraphQlResponse<serde_json::Value>> {
144        let res = self
145            .inner
146            .post(self.rpc_server().clone())
147            .json(&json)
148            .send()
149            .await?
150            .json::<GraphQlResponse<serde_json::Value>>()
151            .await?;
152        Ok(res)
153    }
154
155    /// Handle pagination filters and return the appropriate values. If limit is
156    /// omitted, it will use the max page size from the service config.
157    pub async fn pagination_filter(
158        &self,
159        pagination_filter: PaginationFilter,
160    ) -> PaginationFilterResponse {
161        let limit = pagination_filter
162            .limit
163            .unwrap_or(self.max_page_size().await.unwrap_or(DEFAULT_ITEMS_PER_PAGE));
164
165        let (after, before, first, last) = match pagination_filter.direction {
166            Direction::Forward => (pagination_filter.cursor, None, Some(limit), None),
167            Direction::Backward => (None, pagination_filter.cursor, None, Some(limit)),
168        };
169        PaginationFilterResponse {
170            after,
171            before,
172            first,
173            last,
174        }
175    }
176
177    /// Lazily fetch the max page size
178    pub async fn max_page_size(&self) -> Result<i32> {
179        self.service_config().await.map(|cfg| cfg.max_page_size)
180    }
181}

188    #[test]
189    fn test_rpc_server() {
190        let mut client = Client::new_mainnet();
191        assert_eq!(client.rpc_server(), &MAINNET_HOST.parse().unwrap());
192        client.set_rpc_server(TESTNET_HOST).unwrap();
193        assert_eq!(client.rpc_server(), &TESTNET_HOST.parse().unwrap());
194        client.set_rpc_server(DEVNET_HOST).unwrap();
195        assert_eq!(client.rpc_server(), &DEVNET_HOST.parse().unwrap());
196        client.set_rpc_server(LOCAL_HOST).unwrap();
197        assert_eq!(client.rpc_server(), &LOCAL_HOST.parse().unwrap());
198
199        assert!(client.set_rpc_server("localhost:9125/graphql").is_ok());
200        assert!(client.set_rpc_server("9125/graphql").is_err());
201    }
