31/// Determines what to wait for after executing a transaction.
32///
33/// Users should almost always use [`WaitForTx::Finalized`] (the default).
34/// The GraphQL client interacts with the indexer, not the fullnode directly.
35/// Using [`WaitForTx::IndexedOnNode`] only guarantees the transaction is
36/// indexed on the fullnode (meaning you can submit transactions that reference
37/// objects created by this transaction), but subsequent queries using the
38/// transaction ID can still fail until the transaction is indexed on the
39/// indexer.
40#[non_exhaustive]
41#[derive(Default)]
42pub enum WaitForTx {
43    /// Indicates that the transaction effects will be usable in subsequent
44    /// transactions (you can reference objects created by this transaction),
45    /// and that the transaction itself is indexed on the fullnode.
46    ///
47    /// **Warning:** This does not guarantee the transaction is indexed on the
48    /// indexer. Since the GraphQL client queries the indexer, subsequent
49    /// queries with this transaction ID may still fail. Prefer
50    /// [`WaitForTx::Finalized`] unless you have a specific reason to use this.
51    IndexedOnNode,
52    /// Indicates that the transaction has been included in a checkpoint, and
53    /// all queries may include it.
54    #[default]
55    Finalized,
56}

59    /// Get a transaction by its digest.
60    pub async fn transaction(&self, digest: Digest) -> Result<Option<SignedTransaction>> {
61        let operation = TransactionBlockQuery::build(TransactionBlockArgs {
62            digest: digest.to_string(),
63        });
64        let response = self.run_query(&operation).await?;
65
66        response
67            .transaction_block
68            .map(TryInto::try_into)
69            .transpose()
70    }

72    /// Get a page of transactions based on the provided filters.
73    pub async fn transactions(
74        &self,
75        filter: impl Into<Option<TransactionsFilter>>,
76        pagination_filter: PaginationFilter,
77    ) -> Result<Page<SignedTransaction>> {
78        let pagination = self.pagination_filter(pagination_filter).await;
79
80        let operation = TransactionBlocksQuery::build(TransactionBlocksQueryArgs {
81            after: pagination.after,
82            before: pagination.before,
83            filter: filter.into(),
84            first: pagination.first,
85            last: pagination.last,
86        });
87
88        let response = self.run_query(&operation).await?;
89
90        let txc = response.transaction_blocks;
91        let page_info = txc.page_info;
92
93        let transactions = txc
94            .nodes
95            .into_iter()
96            .map(|n| n.try_into())
97            .collect::<Result<Vec<_>>>()?;
98        Ok(Page::new(page_info, transactions))
99    }

143    /// Get a transaction's data and effects by its digest.
144    pub async fn transaction_data_effects(
145        &self,
146        digest: Digest,
147    ) -> Result<Option<TransactionDataEffects>> {
148        let operation = TransactionBlockWithEffectsQuery::build(TransactionBlockArgs {
149            digest: digest.to_string(),
150        });
151        let response = self.run_query(&operation).await?;
152
153        match response.transaction_block.map(|tx| (tx.bcs, tx.effects)) {
154            Some((Some(bcs), Some(effects))) => {
155                let bcs = base64ct::Base64::decode_vec(bcs.0.as_str())?;
156                let effects = base64ct::Base64::decode_vec(effects.bcs.unwrap().0.as_str())?;
157                let transaction: SenderSignedTransaction = bcs::from_bytes(&bcs)?;
158                let effects: TransactionEffects = bcs::from_bytes(&effects)?;
159
160                Ok(Some(TransactionDataEffects {
161                    tx: transaction.0,
162                    effects,
163                }))
164            }
165            _ => Ok(None),
166        }
167    }

169    /// Get a page of transactions' data and effects based on the provided
170    /// filters.
171    pub async fn transactions_data_effects(
172        &self,
173        filter: impl Into<Option<TransactionsFilter>>,
174        pagination_filter: PaginationFilter,
175    ) -> Result<Page<TransactionDataEffects>> {
176        let pagination = self.pagination_filter(pagination_filter).await;
177
178        let operation = TransactionBlocksWithEffectsQuery::build(TransactionBlocksQueryArgs {
179            after: pagination.after,
180            before: pagination.before,
181            filter: filter.into(),
182            first: pagination.first,
183            last: pagination.last,
184        });
185
186        let response = self.run_query(&operation).await?;
187
188        let txc = response.transaction_blocks;
189        let page_info = txc.page_info;
190
191        let transactions = {
192            txc.nodes
193                .into_iter()
194                .map(|node| {
195                    let (Some(bcs), Some(effects)) = (node.bcs, node.effects) else {
196                        return Err(Error::empty_response_error());
197                    };
198                    let bcs = base64ct::Base64::decode_vec(bcs.0.as_str())?;
199                    let effects =
200                        base64ct::Base64::decode_vec(effects.bcs.as_ref().unwrap().0.as_str())?;
201                    let transaction: SenderSignedTransaction = bcs::from_bytes(&bcs)?;
202                    let effects: TransactionEffects = bcs::from_bytes(&effects)?;
203
204                    Ok(TransactionDataEffects {
205                        tx: transaction.0,
206                        effects,
207                    })
208                })
209                .collect::<Result<Vec<_>>>()?
210        };
211
212        Ok(Page::new(page_info, transactions))
213    }

242    /// Execute a transaction.
243    pub async fn execute_tx(
244        &self,
245        signatures: &[UserSignature],
246        tx: &Transaction,
247        wait_for: impl Into<Option<WaitForTx>>,
248    ) -> Result<TransactionEffects> {
249        let wait_for = wait_for.into();
250        let operation = ExecuteTransactionQuery::build(ExecuteTransactionArgs {
251            signatures: signatures.iter().map(|s| s.to_base64()).collect(),
252            tx_bytes: base64ct::Base64::encode_string(bcs::to_bytes(tx).unwrap().as_ref()),
253        });
254
255        let response = self.run_query(&operation).await?;
256
257        let result = response.execute_transaction_block;
258        let bcs = base64ct::Base64::decode_vec(result.effects.bcs.0.as_str())?;
259        let effects: TransactionEffects = bcs::from_bytes(&bcs)?;
260
261        if let Some(wait_for) = wait_for {
262            self.wait_for_tx(tx.digest(), wait_for, None).await?;
263        }
264
265        Ok(effects)
266    }

268    /// Returns whether the transaction for the given digest has been indexed
269    /// on the node. This means that it can be queried by its digest and its
270    /// effects will be usable for subsequent transactions. To check for
271    /// full finalization, use [`Self::is_tx_finalized`].
272    pub async fn is_tx_indexed_on_node(&self, digest: Digest) -> Result<bool> {
273        let operation = TransactionBlockIndexedQuery::build(TransactionBlockArgs {
274            digest: digest.to_string(),
275        });
276        Ok(self
277            .run_query(&operation)
278            .await?
279            .is_transaction_indexed_on_node)
280    }
281
282    /// Returns whether the transaction for the given digest has been included
283    /// in a checkpoint (finalized).
284    pub async fn is_tx_finalized(&self, digest: Digest) -> Result<bool> {
285        let operation = TransactionBlockCheckpointQuery::build(TransactionBlockArgs {
286            digest: digest.to_string(),
287        });
288        let response = self.run_query(&operation).await?;
289        if let Some(block) = response.transaction_block
290            && block
291                .effects
292                .as_ref()
293                .and_then(|e| e.checkpoint.as_ref())
294                .is_some()
295        {
296            return Ok(true);
297        }
298        Ok(false)
299    }
300
301    /// Wait for the indexing or finalization of a transaction
302    /// by its digest. An optional timeout can be provided, which, if
303    /// exceeded, will return an error (default 60s).
304    pub async fn wait_for_tx(
305        &self,
306        digest: Digest,
307        wait_for: WaitForTx,
308        timeout: impl Into<Option<Duration>>,
309    ) -> Result<()> {
310        tokio::time::timeout(
311            timeout.into().unwrap_or_else(|| Duration::from_secs(60)),
312            async {
313                let mut interval = tokio::time::interval(tokio::time::Duration::from_millis(100));
314                loop {
315                    interval.tick().await;
316                    if match wait_for {
317                        WaitForTx::IndexedOnNode => self.is_tx_indexed_on_node(digest).await?,
318                        WaitForTx::Finalized => self.is_tx_finalized(digest).await?,
319                    } {
320                        break Ok(());
321                    }
322                }
323            },
324        )
325        .await
326        .map_err(|e| Error::from_error(Kind::Other, e))?
327    }
