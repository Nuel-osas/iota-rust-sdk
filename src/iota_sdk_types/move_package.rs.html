6/// Rust representation of upgrade policy constants in `iota::package`.
7#[repr(u8)]
8#[derive(Debug, Clone, Copy, PartialEq, Eq, strum::Display)]
9#[strum(serialize_all = "SCREAMING_SNAKE_CASE")]
10#[non_exhaustive]
11pub enum UpgradePolicy {
12    /// The least restrictive policy. Permits changes to all function
13    /// implementations, the removal of ability constraints on generic type
14    /// parameters in function signatures, and modifications to private,
15    /// public(friend), and entry function signatures. However, public function
16    /// signatures and existing types cannot be changed.
17    Compatible = 0,
18    /// Allows adding new functionalities (e.g., new public functions or
19    /// structs) but restricts changes to existing functionalities.
20    Additive = 128,
21    /// Limits modifications to the package’s dependencies only.
22    DepOnly = 192,
23}
24
25impl UpgradePolicy {
26    pub const COMPATIBLE: u8 = Self::Compatible as u8;
27    pub const ADDITIVE: u8 = Self::Additive as u8;
28    pub const DEP_ONLY: u8 = Self::DepOnly as u8;
29
30    pub fn is_valid_policy(policy: &u8) -> bool {
31        Self::try_from(*policy).is_ok()
32    }
33}
34
35impl TryFrom<u8> for UpgradePolicy {
36    type Error = ();
37    fn try_from(value: u8) -> Result<Self, Self::Error> {
38        match value {
39            x if x == Self::Compatible as u8 => Ok(Self::Compatible),
40            x if x == Self::Additive as u8 => Ok(Self::Additive),
41            x if x == Self::DepOnly as u8 => Ok(Self::DepOnly),
42            _ => Err(()),
43        }
44    }
45}

47/// Type corresponding to the output of `iota move build
48/// --dump-bytecode-as-base64`
49#[derive(Clone, Debug)]
50#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
51pub struct MovePackageData {
52    /// The package modules as a series of bytes
53    #[cfg_attr(feature = "serde", serde(with = "serialization::modules"))]
54    pub modules: Vec<Vec<u8>>,
55    /// The package dependencies, specified by their object IDs.
56    pub dependencies: Vec<ObjectId>,
57    /// The package digest.
58    #[cfg_attr(feature = "serde", serde(with = "serialization::digest"))]
59    pub digest: Digest,
60}
61
62impl MovePackageData {
63    #[cfg(feature = "hash")]
64    pub fn new(modules: Vec<Vec<u8>>, dependencies: Vec<ObjectId>) -> Self {
65        use crate::hash::Hasher;
66        let mut components = dependencies
67            .iter()
68            .map(|o| o.into_inner())
69            .chain(modules.iter().map(|module| {
70                let mut hasher = Hasher::new();
71                hasher.update(module);
72                hasher.finalize().into_inner()
73            }))
74            .collect::<Vec<_>>();
75
76        // Sort so the order of the modules and the order of the dependencies
77        // does not matter.
78        components.sort();
79
80        let mut hasher = Hasher::new();
81        for c in components {
82            hasher.update(c);
83        }
84
85        Self {
86            modules,
87            dependencies,
88            digest: Digest::from(hasher.finalize().into_inner()),
89        }
90    }
91}

93#[cfg(feature = "serde")]
94mod serialization {
95    use base64ct::Encoding;
96    use serde::{Deserialize, Deserializer, Serialize, Serializer};
97
98    use super::*;
99
100    impl MovePackageData {
101        pub fn to_base64(&self) -> String {
102            base64ct::Base64::encode_string(&bcs::to_bytes(self).expect("bcs encoding failed"))
103        }
104
105        pub fn from_base64(base64: &str) -> Result<Self, bcs::Error> {
106            use serde::de::Error;
107            bcs::from_bytes(&base64ct::Base64::decode_vec(base64).map_err(bcs::Error::custom)?)
108        }
109    }
110
111    pub mod modules {
112        use super::*;
113
114        pub fn serialize<S: Serializer>(
115            value: &[Vec<u8>],
116            serializer: S,
117        ) -> Result<S::Ok, S::Error> {
118            value
119                .iter()
120                .map(|v| base64ct::Base64::encode_string(v))
121                .collect::<Vec<_>>()
122                .serialize(serializer)
123        }
124
125        pub fn deserialize<'de, D>(deserializer: D) -> Result<Vec<Vec<u8>>, D::Error>
126        where
127            D: Deserializer<'de>,
128        {
129            let bcs = Vec::<String>::deserialize(deserializer)?;
130            bcs.into_iter()
131                .map(|s| base64ct::Base64::decode_vec(&s).map_err(serde::de::Error::custom))
132                .collect()
133        }
134    }
135
136    pub mod digest {
137        use super::*;
138
139        pub fn serialize<S: Serializer>(value: &Digest, serializer: S) -> Result<S::Ok, S::Error> {
140            value.as_bytes().serialize(serializer)
141        }
142
143        pub fn deserialize<'de, D>(deserializer: D) -> Result<Digest, D::Error>
144        where
145            D: Deserializer<'de>,
146        {
147            let bytes = Vec::<u8>::deserialize(deserializer)?;
148            Digest::from_bytes(bytes).map_err(|e| serde::de::Error::custom(format!("{e}")))
149        }
150    }
151}
