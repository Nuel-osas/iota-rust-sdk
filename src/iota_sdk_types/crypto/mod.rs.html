60/// Utility for calculating base64 encoding lengths.
61///
62/// In the Base64 encoding each character is used to represent 6 bits (log2(64)
63/// = 6). This means that 4 characters are used to represent 4*6 = 24 bits = 3
64/// bytes. So you need 4*(`n`/3) characters in order to represent `n` bytes, and
65/// this needs to be rounded up to a multiple of 4. The number of unused padding
66/// characters resulting from the rounding will be 0, 1, 2, or 3.
67const fn base64_encoded_length(len: usize) -> usize {
68    ((4 * len / 3) + 3) & !3
69}
70
71macro_rules! impl_base64_helper {
72    ($base:ident, $display:ident, $fromstr:ident, $test_module:ident, $array_length:literal) => {
73        #[allow(unused)]
74        struct $base;
75
76        impl $base {
77            const LENGTH: usize = $array_length;
78            #[allow(unused)]
79            const ENCODED_LENGTH: usize = base64_encoded_length(Self::LENGTH);
80        }
81
82        #[allow(unused)]
83        struct $display<'a>(&'a [u8; $base::LENGTH]);
84
85        impl<'a> std::fmt::Display for $display<'a> {
86            fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
87                let mut buf = [0; $base::ENCODED_LENGTH];
88                let encoded =
89                    <base64ct::Base64 as base64ct::Encoding>::encode(self.0, &mut buf).unwrap();
90                f.write_str(encoded)
91            }
92        }
93
94        #[allow(unused)]
95        #[derive(Debug, PartialEq)]
96        #[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
97        struct $fromstr([u8; $base::LENGTH]);
98
99        impl std::str::FromStr for $fromstr {
100            type Err = base64ct::Error;
101
102            fn from_str(s: &str) -> Result<Self, Self::Err> {
103                let mut buf = [0; $base::LENGTH];
104                let decoded = <base64ct::Base64 as base64ct::Encoding>::decode(s, &mut buf)?;
105                assert_eq!(decoded.len(), $base::LENGTH);
106                Ok(Self(buf))
107            }
108        }
109
110        #[cfg(feature = "serde")]
111        #[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
112        impl serde_with::SerializeAs<[u8; Self::LENGTH]> for $base {
113            fn serialize_as<S>(
114                source: &[u8; Self::LENGTH],
115                serializer: S,
116            ) -> Result<S::Ok, S::Error>
117            where
118                S: serde::Serializer,
119            {
120                let display = $display(source);
121                serde_with::DisplayFromStr::serialize_as(&display, serializer)
122            }
123        }
124
125        #[cfg(feature = "serde")]
126        #[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
127        impl<'de> serde_with::DeserializeAs<'de, [u8; Self::LENGTH]> for $base {
128            fn deserialize_as<D>(deserializer: D) -> Result<[u8; Self::LENGTH], D::Error>
129            where
130                D: serde::Deserializer<'de>,
131            {
132                let array: $fromstr = serde_with::DisplayFromStr::deserialize_as(deserializer)?;
133                Ok(array.0)
134            }
135        }
136
137        #[cfg(test)]
138        mod $test_module {
139            use test_strategy::proptest;
140
141            use super::{$display, $fromstr};
142
143            #[proptest]
144            fn roundtrip_display_fromstr(array: $fromstr) {
145                let s = $display(&array.0).to_string();
146                let a = s.parse::<$fromstr>().unwrap();
147                assert_eq!(array, a);
148            }
149        }
150    };
151}

160pub trait PublicKeyExt: Sized {
161    type FromBytesErr;
162
163    /// Returns the public key as bytes.
164    fn as_bytes(&self) -> &[u8];
165
166    /// Tries to create a PublicKey from bytes.
167    fn from_bytes<T: AsRef<[u8]>>(bytes: T) -> Result<Self, Self::FromBytesErr>;
168
169    /// Returns the signature scheme for this public key.
170    fn scheme(&self) -> SignatureScheme;
171
172    /// Returns the bytes with signature scheme flag prepended
173    fn to_flagged_bytes(&self) -> Vec<u8> {
174        let key_bytes = self.as_bytes();
175        let mut bytes = Vec::with_capacity(1 + key_bytes.len());
176        bytes.push(self.scheme().to_u8());
177        bytes.extend_from_slice(key_bytes);
178        bytes
179    }
180}
