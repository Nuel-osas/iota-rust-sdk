4use crate::{Address, Input, ObjectId, ObjectReference, TypeTag};

6/// MoveAuthenticator is a signature variant that enables a method of
7/// authentication through Move code. This type represents the data received
8/// by the Move authenticate function during the Account Abstraction
9/// authentication flow.
10#[derive(Debug, Clone, PartialEq, Eq)]
11#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
12pub struct MoveAuthenticator {
13    /// Input objects or primitive values
14    call_args: Vec<Input>,
15    /// Type arguments for the Move authenticate function
16    type_args: Vec<TypeTag>,
17    /// The object that is authenticated. Represents the account being the
18    /// sender of the transaction.
19    object_to_authenticate: Input,
20}
21
22impl MoveAuthenticator {
23    /// Create a new move authenticator from an immutable object.
24    pub fn new_immutable(
25        call_args: Vec<Input>,
26        type_args: Vec<TypeTag>,
27        object_to_authenticate: ObjectReference,
28    ) -> Self {
29        Self {
30            call_args,
31            type_args,
32            object_to_authenticate: Input::ImmutableOrOwned(object_to_authenticate),
33        }
34    }
35
36    /// Create a new move authenticator from a shared object.
37    pub fn new_shared(
38        call_args: Vec<Input>,
39        type_args: Vec<TypeTag>,
40        object_to_authenticate: ObjectId,
41        initial_shared_version: u64,
42    ) -> Self {
43        Self {
44            call_args,
45            type_args,
46            object_to_authenticate: Input::Shared {
47                object_id: object_to_authenticate,
48                initial_shared_version,
49                mutable: false,
50            },
51        }
52    }
53
54    pub fn address(&self) -> Address {
55        match self.object_to_authenticate {
56            Input::ImmutableOrOwned(ObjectReference { object_id, .. })
57            | Input::Shared { object_id, .. } => object_id.into(),
58            _ => unreachable!(),
59        }
60    }
61
62    pub fn call_args(&self) -> &[Input] {
63        &self.call_args
64    }
65
66    pub fn type_args(&self) -> &[TypeTag] {
67        &self.type_args
68    }
69
70    pub fn object_to_authenticate(&self) -> &Input {
71        &self.object_to_authenticate
72    }
73}

75#[cfg(feature = "serde")]
76#[cfg_attr(doc_cfg, doc(cfg(feature = "serde")))]
77mod serialization {
78    use std::borrow::Cow;
79
80    use serde::{Deserialize, Deserializer, Serialize, Serializer};
81    use serde_with::{Bytes, DeserializeAs};
82
83    use super::*;
84    use crate::{SignatureScheme, crypto::SignatureFromBytesError};
85
86    impl MoveAuthenticator {
87        pub fn from_serialized_bytes(
88            bytes: impl AsRef<[u8]>,
89        ) -> Result<Self, SignatureFromBytesError> {
90            let bytes = bytes.as_ref();
91            let flag =
92                SignatureScheme::from_byte(*bytes.first().ok_or_else(|| {
93                    SignatureFromBytesError::new("missing signature scheme flag")
94                })?)
95                .map_err(SignatureFromBytesError::new)?;
96            if flag != SignatureScheme::MoveAuthenticator {
97                return Err(SignatureFromBytesError::new(
98                    "invalid move authenticator flag",
99                ));
100            }
101            let bcs_bytes = &bytes[1..];
102
103            let auth = bcs::from_bytes::<Authenticator>(bcs_bytes)
104                .map_err(SignatureFromBytesError::new)?;
105
106            Ok(Self {
107                call_args: auth.call_args,
108                type_args: auth.type_args,
109                object_to_authenticate: auth.object_to_authenticate,
110            })
111        }
112
113        pub fn to_bytes(&self) -> Vec<u8> {
114            let as_bytes = bcs::to_bytes(&AuthenticatorRef {
115                call_args: &self.call_args,
116                type_args: &self.type_args,
117                object_to_authenticate: &self.object_to_authenticate,
118            })
119            .expect("BCS serialization should not fail");
120            let mut bytes = Vec::with_capacity(1 + as_bytes.len());
121            bytes.push(SignatureScheme::MoveAuthenticator as u8);
122            bytes.extend(as_bytes);
123            bytes
124        }
125    }
126
127    #[derive(serde::Serialize)]
128    struct AuthenticatorRef<'a> {
129        call_args: &'a Vec<Input>,
130        type_args: &'a Vec<TypeTag>,
131        object_to_authenticate: &'a Input,
132    }
133
134    #[derive(serde::Deserialize)]
135    #[serde(rename = "MoveAuthenticator")]
136    #[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
137    struct Authenticator {
138        call_args: Vec<Input>,
139        #[cfg_attr(feature = "schemars", schemars(with = "Vec<String>"))]
140        type_args: Vec<TypeTag>,
141        object_to_authenticate: Input,
142    }
143
144    #[cfg(feature = "schemars")]
145    impl schemars::JsonSchema for MoveAuthenticator {
146        fn schema_name() -> String {
147            Authenticator::schema_name()
148        }
149
150        fn json_schema(
151            generator: &mut schemars::r#gen::SchemaGenerator,
152        ) -> schemars::schema::Schema {
153            Authenticator::json_schema(generator)
154        }
155    }
156
157    impl Serialize for MoveAuthenticator {
158        fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
159        where
160            S: Serializer,
161        {
162            if serializer.is_human_readable() {
163                let authenticator_ref = AuthenticatorRef {
164                    call_args: &self.call_args,
165                    type_args: &self.type_args,
166                    object_to_authenticate: &self.object_to_authenticate,
167                };
168
169                authenticator_ref.serialize(serializer)
170            } else {
171                let bytes = self.to_bytes();
172                serializer.serialize_bytes(&bytes)
173            }
174        }
175    }
176
177    impl<'de> Deserialize<'de> for MoveAuthenticator {
178        fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
179        where
180            D: Deserializer<'de>,
181        {
182            if deserializer.is_human_readable() {
183                let Authenticator {
184                    call_args,
185                    type_args,
186                    object_to_authenticate,
187                } = Authenticator::deserialize(deserializer)?;
188                Ok(Self {
189                    call_args,
190                    type_args,
191                    object_to_authenticate,
192                })
193            } else {
194                let bytes: Cow<'de, [u8]> = Bytes::deserialize_as(deserializer)?;
195                Self::from_serialized_bytes(bytes)
196            }
197            .map_err(serde::de::Error::custom)
198        }
199    }
