4use std::str::FromStr;
5
6use crate::{ObjectId, address::Address, iota_names::error::IotaNamesError};
7
8#[derive(Clone, Debug, Eq, PartialEq)]
9#[cfg_attr(
10    feature = "serde",
11    derive(serde::Serialize, serde::Deserialize),
12    serde(rename_all = "kebab-case")
13)]
14pub struct IotaNamesConfig {
15    /// Address of the `iota_names` package.
16    pub package_address: Address,
17    /// ID of the `IotaNames` object.
18    pub object_id: ObjectId,
19    /// Address of the `payments` package.
20    pub payments_package_address: Address,
21    /// ID of the registry table.
22    pub registry_id: ObjectId,
23    /// ID of the reverse registry table.
24    pub reverse_registry_id: ObjectId,
25}
26
27impl Default for IotaNamesConfig {
28    fn default() -> Self {
29        Self::mainnet()
30    }
31}
32
33impl IotaNamesConfig {
34    pub fn new(
35        package_address: Address,
36        object_id: ObjectId,
37        payments_package_address: Address,
38        registry_id: ObjectId,
39        reverse_registry_id: ObjectId,
40    ) -> Self {
41        Self {
42            package_address,
43            object_id,
44            payments_package_address,
45            registry_id,
46            reverse_registry_id,
47        }
48    }
49
50    pub fn from_env() -> Result<Self, IotaNamesError> {
51        Ok(Self::new(
52            std::env::var("IOTA_NAMES_PACKAGE_ADDRESS")?.parse()?,
53            std::env::var("IOTA_NAMES_OBJECT_ID")?.parse()?,
54            std::env::var("IOTA_NAMES_PAYMENTS_PACKAGE_ADDRESS")?.parse()?,
55            std::env::var("IOTA_NAMES_REGISTRY_ID")?.parse()?,
56            std::env::var("IOTA_NAMES_REVERSE_REGISTRY_ID")?.parse()?,
57        ))
58    }
59

67            "0x53d3d37f00949a1baad95fa4fca0b3d0d70ff6121be316f9e46d37c2d29c71eb";
68        const REGISTRY_ID: &str =
69            "0xa773cef7d762871354f6ae19ad174dfb1153d2d247c4886ada0b5330b9543b57";
70        const REVERSE_REGISTRY_ID: &str =
71            "0x18fa62ab8b0ab95ae61088082bd5db796863016fda8f3205b1ea7d13b1792317";

80            package_address,
81            object_id,
82            payments_package_address,
83            registry_id,
84            reverse_registry_id,
85        )
86    }
