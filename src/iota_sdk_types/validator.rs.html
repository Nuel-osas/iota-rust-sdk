8/// The Validator Set for a particular epoch.
9///
10/// # BCS
11///
12/// The BCS serialized form for this type is defined by the following ABNF:
13///
14/// ```text
15/// validator-committee = u64 ; epoch
16///                       (vector validator-committee-member)
17/// ```
18#[derive(Clone, Debug, PartialEq, Eq)]
19#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
20#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
21#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
22pub struct ValidatorCommittee {
23    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
24    #[cfg_attr(feature = "schemars", schemars(with = "crate::_schemars::U64"))]
25    pub epoch: EpochId,
26    pub members: Vec<ValidatorCommitteeMember>,
27}
28
29/// A member of a Validator Committee
30///
31/// # BCS
32///
33/// The BCS serialized form for this type is defined by the following ABNF:
34///
35/// ```text
36/// validator-committee-member = bls-public-key
37///                              u64 ; stake
38/// ```
39#[derive(Clone, Debug, PartialEq, Eq)]
40#[cfg_attr(
41    feature = "serde",
42    derive(serde::Serialize, serde::Deserialize),
43    serde(rename_all = "camelCase")
44)]
45#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
46#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
47pub struct ValidatorCommitteeMember {
48    #[cfg_attr(feature = "serde", serde(with = "ValidatorPublicKeySerialization"))]
49    #[cfg_attr(feature = "schemars", schemars(with = "Bls12381PublicKey"))]
50    pub public_key: Bls12381PublicKey,
51    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
52    #[cfg_attr(feature = "schemars", schemars(with = "crate::_schemars::U64"))]
53    pub stake: StakeUnit,
54}

56/// An aggregated signature from multiple Validators.
57///
58/// # BCS
59///
60/// The BCS serialized form for this type is defined by the following ABNF:
61///
62/// ```text
63/// validator-aggregated-signature = u64               ; epoch
64///                                  bls-signature
65///                                  roaring-bitmap
66/// roaring-bitmap = bytes  ; where the contents of the bytes are valid
67///                         ; according to the serialized spec for
68///                         ; roaring bitmaps
69/// ```
70///
71/// See [here](https://github.com/RoaringBitmap/RoaringFormatSpec) for the specification for the
72/// serialized format of RoaringBitmaps.
73#[derive(Clone, Debug, PartialEq)]
74#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
75#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
76#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
77pub struct ValidatorAggregatedSignature {
78    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
79    #[cfg_attr(feature = "schemars", schemars(with = "crate::_schemars::U64"))]
80    pub epoch: EpochId,
81    pub signature: Bls12381Signature,
82    #[cfg_attr(feature = "serde", serde(with = "RoaringBitMapSerialization"))]
83    #[cfg_attr(feature = "schemars", schemars(with = "crate::_schemars::Base64"))]
84    #[cfg_attr(
85        feature = "proptest",
86        strategy(proptest::strategy::Just(roaring::RoaringBitmap::default()))
87    )]
88    pub bitmap: roaring::RoaringBitmap,
89}
90
91#[cfg(feature = "serde")]
92type RoaringBitMapSerialization = ::serde_with::As<
93    ::serde_with::IfIsHumanReadable<
94        crate::_serde::Base64RoaringBitmap,
95        crate::_serde::BinaryRoaringBitmap,
96    >,
97>;
98
99// Similar to Digest...unfortunately validator's public key material is
100// serialized with the length (96) prefixed
101#[cfg(feature = "serde")]
102type ValidatorPublicKeySerialization = ::serde_with::As<
103    ::serde_with::IfIsHumanReadable<::serde_with::DisplayFromStr, BinaryValidatorPublicKey>,
104>;

106#[cfg(feature = "serde")]
107struct BinaryValidatorPublicKey;

109#[cfg(feature = "serde")]
110impl serde_with::SerializeAs<Bls12381PublicKey> for BinaryValidatorPublicKey {
111    fn serialize_as<S>(source: &Bls12381PublicKey, serializer: S) -> Result<S::Ok, S::Error>
112    where
113        S: serde::Serializer,
114    {
115        ::serde_with::Bytes::serialize_as(source.inner(), serializer)
116    }
117}
118
119#[cfg(feature = "serde")]
120impl<'de> serde_with::DeserializeAs<'de, Bls12381PublicKey> for BinaryValidatorPublicKey {
121    fn deserialize_as<D>(deserializer: D) -> Result<Bls12381PublicKey, D::Error>
122    where
123        D: serde::Deserializer<'de>,
124    {
125        let bytes: [u8; Bls12381PublicKey::LENGTH] =
126            ::serde_with::Bytes::deserialize_as(deserializer)?;
127        Ok(Bls12381PublicKey::new(bytes))
128    }
129}

131/// A signature from a Validator
132///
133/// # BCS
134///
135/// The BCS serialized form for this type is defined by the following ABNF:
136///
137/// ```text
138/// validator-signature = u64               ; epoch
139///                       bls-public-key
140///                       bls-signature
141/// ```
142#[derive(Clone, Debug, PartialEq, Eq)]
143#[cfg_attr(
144    feature = "serde",
145    derive(serde::Serialize, serde::Deserialize),
146    serde(rename_all = "camelCase")
147)]
148#[cfg_attr(feature = "schemars", derive(schemars::JsonSchema))]
149#[cfg_attr(feature = "proptest", derive(test_strategy::Arbitrary))]
150pub struct ValidatorSignature {
151    #[cfg_attr(feature = "serde", serde(with = "crate::_serde::ReadableDisplay"))]
152    #[cfg_attr(feature = "schemars", schemars(with = "crate::_schemars::U64"))]
153    pub epoch: EpochId,
154    #[cfg_attr(feature = "serde", serde(with = "ValidatorPublicKeySerialization"))]
155    #[cfg_attr(feature = "schemars", schemars(with = "Bls12381PublicKey"))]
156    pub public_key: Bls12381PublicKey,
157    pub signature: Bls12381Signature,
158}
