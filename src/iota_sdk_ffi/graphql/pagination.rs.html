18macro_rules! define_paged_record {
19    ($id:ident, $type_:ty) => {
20        /// A page of items returned by the GraphQL server.
21        #[derive(uniffi::Record)]
22        pub struct $id {
23            /// Information about the page, such as the cursor and whether there are
24            /// more pages.
25            pub page_info: PageInfo,
26            /// The data returned by the server.
27            pub data: Vec<$type_>,
28        }
29
30        impl From<iota_sdk::graphql_client::pagination::Page<$type_>> for $id {
31            fn from(value: iota_sdk::graphql_client::pagination::Page<$type_>) -> Self {
32                Self {
33                    page_info: value.page_info.into(),
34                    data: value.data,
35                }
36            }
37        }
38    };
39}
40
41define_paged_record!(SignedTransactionPage, SignedTransaction);
42define_paged_record!(TransactionDataEffectsPage, TransactionDataEffects);
43define_paged_record!(DynamicFieldOutputPage, DynamicFieldOutput);
44define_paged_record!(EventPage, Event);
45define_paged_record!(EpochPage, Epoch);
46define_paged_record!(ValidatorPage, Validator);

48macro_rules! define_paged_object {
49    ($id:ident, $type_:ty) => {
50        /// A page of items returned by the GraphQL server.
51        #[derive(uniffi::Record)]
52        pub struct $id {
53            /// Information about the page, such as the cursor and whether there are
54            /// more pages.
55            pub page_info: PageInfo,
56            /// The data returned by the server.
57            pub data: Vec<std::sync::Arc<$type_>>,
58        }
59
60        impl From<iota_sdk::graphql_client::pagination::Page<$type_>> for $id {
61            fn from(value: iota_sdk::graphql_client::pagination::Page<$type_>) -> Self {
62                Self {
63                    page_info: value.page_info.into(),
64                    data: value
65                        .data
66                        .into_iter()
67                        .map(Into::into)
68                        .map(std::sync::Arc::new)
69                        .collect(),
70                }
71            }
72        }
73    };
74}
75
76define_paged_object!(CoinPage, Coin);
77define_paged_object!(ObjectPage, Object);
78define_paged_object!(TransactionEffectsPage, TransactionEffects);
79define_paged_object!(MovePackagePage, MovePackage);
80define_paged_object!(CheckpointSummaryPage, CheckpointSummary);
81define_paged_object!(NameRegistrationPage, NameRegistration);
