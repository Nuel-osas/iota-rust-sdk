4//! Events API implementation.
5
6use iota_sdk::graphql_client::pagination::PaginationFilter;
7
8use crate::{
9    error::Result,
10    graphql::{client::GraphQLClient, pagination::EventPage, query_types::EventFilter},
11};
12
13#[uniffi::export(async_runtime = "tokio")]
14impl GraphQLClient {
15    /// Return a page of tuple (event, transaction digest) based on the
16    /// (optional) event filter.
17    #[uniffi::method(default(pagination_filter = None, filter = None))]
18    pub async fn events(
19        &self,
20        filter: Option<EventFilter>,
21        pagination_filter: Option<PaginationFilter>,
22    ) -> Result<EventPage> {
23        Ok(self
24            .0
25            .read()
26            .await
27            .events(
28                filter.map(|f| f.into()),
29                pagination_filter.unwrap_or_default(),
30            )
31            .await?
32            .map(Into::into)
33            .into())
34    }
35}
