4//! IOTA Names API implementation.
5
6use std::sync::Arc;
7
8use iota_sdk::{graphql_client::pagination::PaginationFilter, types::iota_names::NameFormat};
9
10use crate::{
11    error::Result,
12    graphql::{client::GraphQLClient, pagination::NameRegistrationPage},
13    types::{address::Address, iota_names::Name},
14};
15
16#[uniffi::export(async_runtime = "tokio")]
17impl GraphQLClient {
18    /// Return the resolved address for the given name.
19    pub async fn iota_names_lookup(&self, name: &str) -> Result<Option<Arc<Address>>> {
20        Ok(self
21            .0
22            .read()
23            .await
24            .iota_names_lookup(name)
25            .await?
26            .map(Into::into)
27            .map(Arc::new))
28    }
29
30    /// Find all registration NFTs for the given address.
31    pub async fn iota_names_registrations(
32        &self,
33        address: &Address,
34        pagination_filter: PaginationFilter,
35    ) -> Result<NameRegistrationPage> {
36        Ok(self
37            .0
38            .read()
39            .await
40            .iota_names_registrations(**address, pagination_filter)
41            .await?
42            .map(Into::into)
43            .into())
44    }
45
46    /// Get the default name pointing to this address, if one exists.
47    pub async fn iota_names_default_name(
48        &self,
49        address: &Address,
50        format: Option<NameFormat>,
51    ) -> Result<Option<Arc<Name>>> {
52        Ok(self
53            .0
54            .read()
55            .await
56            .iota_names_default_name(**address, format)
57            .await?
58            .map(Into::into)
59            .map(Arc::new))
60    }
61}
