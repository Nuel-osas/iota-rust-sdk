4//! Dynamic fields API implementation.
5
6use iota_sdk::graphql_client::pagination::PaginationFilter;
7
8use crate::{
9    error::Result,
10    graphql::{
11        client::GraphQLClient, pagination::DynamicFieldOutputPage, query_types::DynamicFieldOutput,
12    },
13    types::{address::Address, type_tag::TypeTag},
14};
15
16#[uniffi::export(async_runtime = "tokio")]
17impl GraphQLClient {
18    /// Access a dynamic field on an object using its name. Names are arbitrary
19    /// Move values whose type have copy, drop, and store, and are specified
20    /// using their type, and their BCS contents, Base64 encoded.
21    ///
22    /// The `name` argument is a json serialized type.
23    ///
24    /// This returns `DynamicFieldOutput` which contains the name, the value
25    /// as json, and object.
26    pub async fn dynamic_field(
27        &self,
28        address: &Address,
29        type_tag: &TypeTag,
30        name: serde_json::Value,
31    ) -> Result<Option<DynamicFieldOutput>> {
32        Ok(self
33            .0
34            .read()
35            .await
36            .dynamic_field(**address, type_tag.0.clone(), name)
37            .await?
38            .map(Into::into))
39    }
40
41    /// Access a dynamic object field on an object using its name. Names are
42    /// arbitrary Move values whose type have copy, drop, and store, and are
43    /// specified using their type, and their BCS contents, Base64 encoded.
44    ///
45    /// The `name` argument is a json serialized type.
46    ///
47    /// This returns `DynamicFieldOutput` which contains the name, the value
48    /// as json, and object.
49    pub async fn dynamic_object_field(
50        &self,
51        address: &Address,
52        type_tag: &TypeTag,
53        name: serde_json::Value,
54    ) -> Result<Option<DynamicFieldOutput>> {
55        Ok(self
56            .0
57            .read()
58            .await
59            .dynamic_object_field(**address, type_tag.0.clone(), name)
60            .await?
61            .map(Into::into))
62    }
63
64    /// Get a page of dynamic fields for the provided address. Note that this
65    /// will also fetch dynamic fields on wrapped objects.
66    ///
67    /// This returns a page of `DynamicFieldOutput`s.
68    #[uniffi::method(default(pagination_filter = None))]
69    pub async fn dynamic_fields(
70        &self,
71        address: &Address,
72        pagination_filter: Option<PaginationFilter>,
73    ) -> Result<DynamicFieldOutputPage> {
74        Ok(self
75            .0
76            .read()
77            .await
78            .dynamic_fields(**address, pagination_filter.unwrap_or_default())
79            .await?
80            .map(Into::into)
81            .into())
82    }
83}
