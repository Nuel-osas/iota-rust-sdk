195#[derive(uniffi::Record)]
196pub struct Epoch {
197    /// The epoch's id as a sequence number that starts at 0 and is incremented
198    /// by one at every epoch change.
199    pub epoch_id: u64,
200    /// The storage fees paid for transactions executed during the epoch.
201    #[uniffi(default = None)]
202    pub fund_inflow: Option<String>,
203    /// The storage fee rebates paid to users who deleted the data associated
204    /// with past transactions.
205    #[uniffi(default = None)]
206    pub fund_outflow: Option<String>,
207    /// The storage fund available in this epoch.
208    /// This fund is used to redistribute storage fees from past transactions
209    /// to future validators.
210    #[uniffi(default = None)]
211    pub fund_size: Option<String>,
212    /// A commitment by the committee at the end of epoch on the contents of the
213    /// live object set at that time. This can be used to verify state
214    /// snapshots.
215    #[uniffi(default = None)]
216    pub live_object_set_digest: Option<String>,
217    /// The difference between the fund inflow and outflow, representing
218    /// the net amount of storage fees accumulated in this epoch.
219    #[uniffi(default = None)]
220    pub net_inflow: Option<String>,
221    /// The epoch's corresponding protocol configuration, including the feature
222    /// flags and the configuration options.
223    #[uniffi(default = None)]
224    pub protocol_configs: Option<ProtocolConfigs>,
225    /// The minimum gas price that a quorum of validators are guaranteed to sign
226    /// a transaction for.
227    #[uniffi(default = None)]
228    pub reference_gas_price: Option<String>,
229    /// The epoch's starting timestamp.
230    pub start_timestamp: u64,
231    /// The epoch's ending timestamp. Note that this is available only on epochs
232    /// that have ended.
233    #[uniffi(default = None)]
234    pub end_timestamp: Option<u64>,
235    /// The value of the `version` field of `0x5`, the
236    /// `0x3::iota::IotaSystemState` object.  This version changes whenever
237    /// the fields contained in the system state object (held in a dynamic
238    /// field attached to `0x5`) change.
239    #[uniffi(default = None)]
240    pub system_state_version: Option<u64>,
241    /// The total number of checkpoints in this epoch.
242    #[uniffi(default = None)]
243    pub total_checkpoints: Option<u64>,
244    /// The total amount of gas fees (in IOTA) that were paid in this epoch.
245    #[uniffi(default = None)]
246    pub total_gas_fees: Option<String>,
247    /// The total IOTA rewarded as stake.
248    #[uniffi(default = None)]
249    pub total_stake_rewards: Option<String>,
250    /// The total number of transaction in this epoch.
251    #[uniffi(default = None)]
252    pub total_transactions: Option<u64>,
253    /// Validator related properties. For active validators, see
254    /// `active_validators` API.
255    /// For epochs other than the current the data provided refer to the start
256    /// of the epoch.
257    #[uniffi(default = None)]
258    pub validator_set: Option<ValidatorSet>,
259}

289                staking_pool_mappings_size: vs.staking_pool_mappings_size,
290                total_stake: vs.total_stake.map(|v| v.0),
291                validator_candidates_size: vs.validator_candidates_size,
292                validator_candidates_id: vs.validator_candidates_id.map(Into::into).map(Arc::new),

576    pub credentials: Option<ValidatorCredentials>,
577    /// Validator's description.
578    #[uniffi(default = None)]
579    pub description: Option<String>,
580    /// Number of exchange rates in the table.
581    #[uniffi(default = None)]
582    pub exchange_rates_size: Option<u64>,
583    /// The reference gas price for this epoch.
584    #[uniffi(default = None)]
585    pub gas_price: Option<u64>,
586    /// Validator's name.
587    #[uniffi(default = None)]
588    pub name: Option<String>,
589    /// Validator's url containing their custom image.
590    #[uniffi(default = None)]
591    pub image_url: Option<String>,
592    /// The proposed next epoch fee for the validator's staking services.
593    #[uniffi(default = None)]

1270    /// Maximum time in milliseconds spent waiting for a response from fullnode
