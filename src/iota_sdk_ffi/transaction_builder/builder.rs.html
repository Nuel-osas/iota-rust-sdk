28/// A builder for creating transactions. Use `finish` to finalize the
29/// transaction data.
30#[derive(Debug, derive_more::From, uniffi::Object)]
31#[uniffi::export(Debug)]
32pub struct TransactionBuilder(RwLock<iota_sdk::transaction_builder::TransactionBuilder<()>>);
33
34impl TransactionBuilder {
35    fn read<F, T>(&self, f: F) -> T
36    where
37        F: FnOnce(&iota_sdk::transaction_builder::TransactionBuilder<()>) -> T,
38    {
39        let lock = self.0.read().expect("error reading from builder");
40        f(&lock)
41    }
42
43    fn write<F, T>(&self, f: F) -> T
44    where
45        F: FnOnce(&mut iota_sdk::transaction_builder::TransactionBuilder<()>) -> T,
46    {
47        let mut lock = self.0.write().expect("error writing to builder");
48        f(&mut lock)
49    }
50}

106    /// Set the gas station sponsor.
107    #[uniffi::method(default(duration = None, headers = None))]
108    pub fn gas_station_sponsor(
109        self: Arc<Self>,
110        url: String,
111        duration: Option<Duration>,
112        headers: Option<HashMap<String, Vec<String>>>,
113    ) -> Arc<Self> {
114        self.write(|builder| {
115            let b = builder.gas_station_sponsor(url.parse().expect("invalid URL"));
116            if let Some(duration) = duration {
117                b.gas_reservation_duration(duration);
118            }
119            if let Some(headers) = headers {
120                for (name, values) in headers {
121                    for value in values {
122                        b.add_gas_station_header(
123                            name.parse().expect("invalid header name"),
124                            value.parse().expect("invalid header value"),
125                        );
126                    }
127                }
128            }
129        });
130        self
131    }

143    /// Call a Move function with the given arguments.
144    #[uniffi::method(default(type_args = [], arguments = [], names = []))]
145    pub fn move_call(
146        self: Arc<Self>,
147        package: &Address,
148        module: &Identifier,
149        function: &Identifier,
150        arguments: Vec<Arc<PTBArgument>>,
151        type_args: Vec<Arc<TypeTag>>,
152        names: Vec<String>,
153    ) -> Arc<Self> {
154        self.write(|builder| {
155            builder
156                .move_call(**package, &module.as_str(), &function.as_str())
157                .arguments(arguments)
158                .type_tags(type_args.into_iter().map(|v| v.0.clone()))
159                .assign(names);
160        });
161        self
162    }

164    /// Send IOTA to a recipient address.
165    ///
166    /// The `amount` parameter specifies the quantity in NANOS, where 1 IOTA
167    /// equals 1_000_000_000 NANOS. That amount is split from the gas coin and
168    /// sent.
169    pub fn send_iota(self: Arc<Self>, recipient: &Address, amount: &PTBArgument) -> Arc<Self> {
170        self.write(|builder| {
171            builder.send_iota(**recipient, amount);
172        });
173        self
174    }

176    /// Transfer some coins to a recipient address. If multiple coins are
177    /// provided then they will be merged.
178    ///
179    /// The `amount` parameter specifies the quantity in NANOS, where 1 IOTA
180    /// equals 1_000_000_000 NANOS.
181    /// If `amount` is provided, that amount is split from the provided coins
182    /// and sent.
183    /// If `amount` is `None`, the entire coins are transferred.
184    ///
185    /// All provided coins must have the same coin type. Mixing coins of
186    /// different types will result in an error.
187    ///
188    /// If you intend to transfer all provided coins to another address in a
189    /// single transaction, consider using
190    /// `TransactionBuilder::transfer_objects()` instead.
191    #[uniffi::method(default(amount = None))]
192    pub fn send_coins(
193        self: Arc<Self>,
194        coins: Vec<Arc<PTBArgument>>,
195        recipient: &Address,
196        amount: Option<Arc<PTBArgument>>,
197    ) -> Arc<Self> {
198        self.write(|builder| {
199            builder.send_coins::<_, &PTBArgument>(coins, **recipient, amount.as_deref());
200        });
201        self
202    }
203

231    /// Merge multiple coins into one.
232    ///
233    /// This method combines the balances of multiple coins of the same coin
234    /// type into a single coin. The `primary_coin` will receive the balances
235    /// from all `consumed_coins`. After merging, the `consumed_coins` will
236    /// be consumed and no longer exist.

238        self: Arc<Self>,
239        primary_coin: &PTBArgument,

242        self.write(|builder| {
243            builder.merge_coins(primary_coin, consumed_coins);
244        });
245        self

248    /// Make a move vector from a list of elements. The elements must all be of
249    /// the type indicated by `type_tag`.
250    pub fn make_move_vec(
251        self: Arc<Self>,
252        elements: Vec<Arc<MoveArg>>,
253        type_tag: &TypeTag,
254        name: String,
255    ) -> Arc<Self> {
256        use iota_sdk::transaction_builder::unresolved::{Command, MakeMoveVector};
257        self.write(|builder| {
258            let cmd = Command::MakeMoveVector(MakeMoveVector {
259                type_: Some(type_tag.0.clone()),
260                elements: elements
261                    .iter()
262                    .map(|e| builder.apply_argument(e.as_ref()))
263                    .collect(),
264            });
265            builder.assigned_command(cmd, name);
266        });
267        self
268    }

270    /// Publish a list of modules with the given dependencies. The result
271    /// assigned to `upgrade_cap_name` is the `0x2::package::UpgradeCap`
272    /// Move type. Note that the upgrade capability needs to be handled
273    /// after this call:
274    ///  - transfer it to the transaction sender or another address
275    ///  - burn it
276    ///  - wrap it for access control
277    ///  - discard the it to make a package immutable
278    ///
279    /// The arguments required for this command are:
280    ///  - `modules`: is the modules' bytecode to be published
281    ///  - `dependencies`: is the list of IDs of the transitive dependencies of
282    ///    the package
283    pub fn publish(
284        self: Arc<Self>,
285        package_data: &MovePackageData,
286        upgrade_cap_name: String,
287    ) -> Arc<Self> {
288        self.write(|builder| {
289            builder
290                .publish(package_data.0.clone())
291                .upgrade_cap(upgrade_cap_name);
292        });
293        self
294    }
295
296    /// Upgrade a Move package.
297    ///
298    ///  - `modules`: is the modules' bytecode for the modules to be published
299    ///  - `dependencies`: is the list of IDs of the transitive dependencies of
300    ///    the package to be upgraded
301    ///  - `package`: is the ID of the current package being upgraded
302    ///  - `ticket`: is the upgrade ticket
303    ///
304    ///  To get the ticket, you have to call the
305    /// `0x2::package::authorize_upgrade` function, and pass the package
306    /// ID, the upgrade policy, and package digest.
307    #[uniffi::method(default(name = None))]
308    pub fn upgrade(
309        self: Arc<Self>,
310        package_id: &ObjectId,
311        package_data: &MovePackageData,
312        upgrade_ticket: &PTBArgument,
313        name: Option<String>,
314    ) -> Arc<Self> {
315        self.write(|builder| {
316            builder
317                .upgrade(**package_id, package_data.0.clone(), upgrade_ticket)
318                .assign(name);
319        });
320        self
321    }

342    /// Convert this builder into a transaction.
343    pub fn finish(&self) -> Result<Transaction> {
344        Ok(Transaction(self.read(|builder| builder.clone().finish())?))
345    }

347    /// Execute the transaction using the gas station and return the JSON
348    /// transaction effects. This will fail unless data is set with the
349    /// `gas_station_sponsor` function.
350    ///
351    /// NOTE: These effects are not necessarily compatible with
352    /// `TransactionEffects`
353    pub async fn execute_with_gas_station(
354        &self,
355        signer: &TransactionSigner,
356    ) -> Result<serde_json::Value> {
357        Ok(self
358            .read(|builder| builder.clone().execute_with_gas_station(signer))
359            .await?)
