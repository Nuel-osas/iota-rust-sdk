4use std::sync::RwLock;
5
6use crate::{
7    error::Result,
8    types::{
9        checkpoint::CheckpointSummary,
10        validator::{ValidatorAggregatedSignature, ValidatorCommittee, ValidatorSignature},
11    },

14#[derive(Debug, derive_more::From, uniffi::Object)]
15#[uniffi::export(Debug)]
16pub struct ValidatorCommitteeSignatureVerifier(
17    pub iota_sdk::crypto::validator::ValidatorCommitteeSignatureVerifier,
18);
19
20#[uniffi::export]
21impl ValidatorCommitteeSignatureVerifier {
22    #[uniffi::constructor]
23    pub fn new(committee: ValidatorCommittee) -> Result<Self> {
24        Ok(Self(
25            iota_sdk::crypto::validator::ValidatorCommitteeSignatureVerifier::new(
26                committee.into(),
27            )?,
28        ))
29    }
30
31    pub fn committee(&self) -> ValidatorCommittee {
32        self.0.committee().clone().into()
33    }
34
35    pub fn verify_checkpoint_summary(
36        &self,
37        summary: &CheckpointSummary,
38        signature: &ValidatorAggregatedSignature,
39    ) -> Result<()> {
40        Ok(self.0.verify_checkpoint_summary(&summary.0, &signature.0)?)
41    }
42
43    pub fn verify(&self, message: &[u8], signature: &ValidatorSignature) -> Result<()> {
44        Ok(iota_sdk::crypto::Verifier::<
45            iota_sdk::types::ValidatorSignature,
46        >::verify(&self.0, message, &signature.0)?)
47    }
48
49    pub fn verify_aggregated(
50        &self,
51        message: &[u8],
52        signature: &ValidatorAggregatedSignature,
53    ) -> Result<()> {
54        Ok(iota_sdk::crypto::Verifier::<
55            iota_sdk::types::ValidatorAggregatedSignature,
56        >::verify(&self.0, message, &signature.0)?)
57    }
58}
59
60#[derive(Debug, derive_more::From, uniffi::Object)]
61#[uniffi::export(Debug)]
62pub struct ValidatorCommitteeSignatureAggregator(
63    pub RwLock<iota_sdk::crypto::validator::ValidatorCommitteeSignatureAggregator>,
64);
65
66#[uniffi::export]
67impl ValidatorCommitteeSignatureAggregator {
68    #[uniffi::constructor]
69    pub fn new_checkpoint_summary(
70        committee: ValidatorCommittee,
71        summary: &CheckpointSummary,
72    ) -> Result<Self> {
73        Ok(Self(
74            iota_sdk::crypto::validator::ValidatorCommitteeSignatureAggregator::new_checkpoint_summary(
75                committee.into(),
76                &summary.0,
77            )?
78            .into(),
79        ))
80    }
81
82    pub fn committee(&self) -> ValidatorCommittee {
83        self.0
84            .read()
85            .expect("failed to read validator committee signature aggregator")
86            .committee()
87            .clone()
88            .into()
89    }
90
91    pub fn add_signature(&self, signature: &ValidatorSignature) -> Result<()> {
92        Ok(self
93            .0
94            .write()
95            .expect("failed to read validator committee signature aggregator")
96            .add_signature(signature.0.clone())?)
97    }
98
99    pub fn finish(&self) -> Result<ValidatorAggregatedSignature> {
100        Ok(self
101            .0
102            .read()
103            .expect("failed to read validator committee signature aggregator")
104            .finish()?
105            .into())
106    }
107}
