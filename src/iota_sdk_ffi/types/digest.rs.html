4use crate::error::Result;

6/// A 32-byte Blake2b256 hash output.
7///
8/// # BCS
9///
10/// A `Digest`'s BCS serialized form is defined by the following:
11///
12/// ```text
13/// digest = %x20 32OCTET
14/// ```
15///
16/// Due to historical reasons, even though a `Digest` has a fixed-length of 32,
17/// IOTA's binary representation of a `Digest` is prefixed with its length
18/// meaning its serialized binary form (in bcs) is 33 bytes long vs a more
19/// compact 32 bytes.
20#[derive(
21    Debug,
22    Hash,
23    PartialEq,
24    Eq,
25    PartialOrd,
26    Ord,
27    derive_more::Display,
28    derive_more::From,
29    derive_more::Deref,
30    uniffi::Object,
31)]
32#[uniffi::export(Debug, Display, Hash, Eq)]
33pub struct Digest(pub iota_sdk::types::Digest);
34
35#[uniffi::export]
36impl Digest {
37    #[uniffi::constructor]
38    pub fn from_bytes(bytes: Vec<u8>) -> Result<Self> {
39        Ok(Self(iota_sdk::types::Digest::from_bytes(bytes)?))
40    }
41
42    #[uniffi::constructor]
43    pub fn from_base58(base58: &str) -> Result<Self> {
44        Ok(Self(iota_sdk::types::Digest::from_base58(base58)?))
45    }
46
47    #[uniffi::constructor]
48    pub fn generate() -> Self {
49        let mut rng = rand::thread_rng();
50        Self(iota_sdk::types::Digest::generate(&mut rng))
51    }
52
53    pub fn to_bytes(&self) -> Vec<u8> {
54        self.0.as_bytes().to_vec()
55    }
56
57    pub fn to_base58(&self) -> String {
58        self.0.to_base58()
59    }
60
61    /// Returns the next digest in byte-increasing order.
62    pub fn next_lexicographical(&self) -> Self {
63        self.0.next_lexicographical().into()
64    }
65}
