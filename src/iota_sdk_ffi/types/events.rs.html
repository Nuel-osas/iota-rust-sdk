16/// An event
17///
18/// # BCS
19///
20/// The BCS serialized form for this type is defined by the following ABNF:
21///
22/// ```text
23/// event = object-id identifier address struct-tag bytes
24/// ```
25#[derive(uniffi::Record)]
26pub struct Event {
27    /// Package id of the top-level function invoked by a MoveCall command which
28    /// triggered this event to be emitted.
29    pub package_id: Arc<ObjectId>,
30    /// Module name of the top-level function invoked by a MoveCall command
31    /// which triggered this event to be emitted.
32    pub module: String,
33    /// Address of the account that sent the transaction where this event was
34    /// emitted.
35    pub sender: Arc<Address>,
36    /// The type of the event emitted
37    pub type_: String,
38    /// BCS serialized bytes of the event
39    pub contents: Vec<u8>,
40    /// UTC timestamp in milliseconds since epoch (1/1/1970)
41    pub timestamp: String,
42    /// Structured contents of a Move value
43    pub data: String,
44    /// Representation of a Move value in JSON
45    pub json: String,
46}
47
48impl From<iota_sdk::graphql_client::query_types::Event> for Event {
49    fn from(value: iota_sdk::graphql_client::query_types::Event) -> Self {
50        let sending_module = value.sending_module.as_ref().unwrap();
51        Self {
52            package_id: Arc::new(ObjectId(iota_sdk::types::ObjectId::from(
53                sending_module.package.address,
54            ))),
55            module: sending_module.name.clone(),
56            sender: Arc::new(Address(value.sender.as_ref().unwrap().address)),
57            type_: value.type_.repr.clone(),
58            contents: base64ct::Base64::decode_vec(&value.bcs.0).unwrap_or_default(),
59            timestamp: value.timestamp.as_ref().unwrap().0.clone(),
60            data: value.data.0.to_string(),
61            json: value.json.to_string(),
62        }
63    }
64}

66impl From<Event> for iota_sdk::types::Event {
67    fn from(value: Event) -> Self {
68        Self {
69            package_id: (**value.package_id),
70            module: Identifier::from_str(&value.module).unwrap(),
71            sender: (**value.sender),
72            type_: StructTag::from_str(&value.type_).unwrap(),
73            contents: value.contents,
74        }
75    }
76}
77
78impl From<Event> for iota_sdk::graphql_client::query_types::Event {
79    fn from(value: Event) -> Self {
80        Self {
81            sending_module: Some(MoveModuleQuery {
82                package: MovePackageQuery {
83                    address: iota_sdk::types::Address::from(**value.package_id),
84                    bcs: None,
85                },
86                name: value.module.clone(),
87            }),
88            sender: Some(GQLAddress {
89                address: (**value.sender),
90            }),
91            type_: MoveType {
92                repr: value.type_.clone(),
93            },
94            bcs: Base64(base64ct::Base64::encode_string(&value.contents)),
95            timestamp: Some(DateTime(value.timestamp.clone())),
96            data: MoveData(serde_json::from_str(&value.data).unwrap_or_default()),
97            json: serde_json::Value::from_str(&value.json).unwrap_or_default(),
98        }
99    }
100}

117/// Events emitted during the successful execution of a transaction
118///
119/// # BCS
120///
121/// The BCS serialized form for this type is defined by the following ABNF:
122///
123/// ```text
124/// transaction-events = vector event
125/// ```
126#[derive(derive_more::From, uniffi::Object)]
127pub struct TransactionEvents(pub iota_sdk::types::TransactionEvents);
128
129#[uniffi::export]
130impl TransactionEvents {
131    #[uniffi::constructor]
132    pub fn new(events: Vec<Event>) -> Self {
133        Self(iota_sdk::types::TransactionEvents(
134            events.into_iter().map(Into::into).collect(),
135        ))
136    }
137
138    pub fn events(&self) -> Vec<Event> {
139        self.0.0.iter().cloned().map(Into::into).collect()
140    }
141
142    pub fn digest(&self) -> Digest {
143        self.0.digest().into()
144    }
145}
