14/// Version 1 of TransactionEffects
15///
16/// # BCS
17///
18/// The BCS serialized form for this type is defined by the following ABNF:
19///
20/// ```text
21/// effects-v1 = execution-status
22///              u64                                ; epoch
23///              gas-cost-summary
24///              digest                             ; transaction digest
25///              (option u32)                       ; gas object index
26///              (option digest)                    ; events digest
27///              (vector digest)                    ; list of transaction dependencies
28///              u64                                ; lamport version
29///              (vector changed-object)
30///              (vector unchanged-shared-object)
31///              (option digest)                    ; auxiliary data digest
32/// ```
33#[derive(uniffi::Record)]
34pub struct TransactionEffectsV1 {
35    /// The status of the execution
36    pub status: ExecutionStatus,
37    /// The epoch when this transaction was executed.
38    pub epoch: u64,
39    /// The gas used by this transaction
40    pub gas_used: GasCostSummary,
41    /// The transaction digest
42    pub transaction_digest: Arc<Digest>,
43    /// The updated gas object reference, as an index into the `changed_objects`
44    /// vector. Having a dedicated field for convenient access.
45    /// System transaction that don't require gas will leave this as None.
46    #[uniffi(default = None)]
47    pub gas_object_index: Option<u32>,
48    /// The digest of the events emitted during execution,
49    /// can be None if the transaction does not emit any event.
50    #[uniffi(default = None)]
51    pub events_digest: Option<Arc<Digest>>,
52    /// The set of transaction digests this transaction depends on.
53    pub dependencies: Vec<Arc<Digest>>,
54    /// The version number of all the written Move objects by this transaction.
55    pub lamport_version: u64,
56    /// Objects whose state are changed in the object store.
57    pub changed_objects: Vec<ChangedObject>,
58    /// Shared objects that are not mutated in this transaction. Unlike owned
59    /// objects, read-only shared objects' version are not committed in the
60    /// transaction, and in order for a node to catch up and execute it
61    /// without consensus sequencing, the version needs to be committed in
62    /// the effects.
63    pub unchanged_shared_objects: Vec<UnchangedSharedObject>,
64    /// Auxiliary data that are not protocol-critical, generated as part of the
65    /// effects but are stored separately. Storing it separately allows us
66    /// to avoid bloating the effects with data that are not critical.
67    /// It also provides more flexibility on the format and type of the data.
68    #[uniffi(default = None)]
69    pub auxiliary_data_digest: Option<Arc<Digest>>,
70}
71
72impl From<iota_sdk::types::TransactionEffectsV1> for TransactionEffectsV1 {
73    fn from(value: iota_sdk::types::TransactionEffectsV1) -> Self {
74        Self {
75            status: value.status.into(),
76            epoch: value.epoch,
77            gas_used: value.gas_used,
78            transaction_digest: Arc::new(value.transaction_digest.into()),
79            gas_object_index: value.gas_object_index,
80            events_digest: value.events_digest.map(Into::into).map(Arc::new),
81            dependencies: value
82                .dependencies
83                .into_iter()
84                .map(Into::into)
85                .map(Arc::new)
86                .collect(),
87            lamport_version: value.lamport_version,
88            changed_objects: value.changed_objects.into_iter().map(Into::into).collect(),
89            unchanged_shared_objects: value
90                .unchanged_shared_objects
91                .into_iter()
92                .map(Into::into)
93                .collect(),
94            auxiliary_data_digest: value.auxiliary_data_digest.map(Into::into).map(Arc::new),
95        }
96    }
97}

117        }
118    }

121/// Input/output state of an object that was changed during execution
122///
123/// # BCS
124///
125/// The BCS serialized form for this type is defined by the following ABNF:
126///
127/// ```text
128/// changed-object = object-id object-in object-out id-operation
129/// ```
130#[derive(uniffi::Record)]
131pub struct ChangedObject {
132    /// Id of the object
133    pub object_id: Arc<ObjectId>,
134    /// State of the object in the store prior to this transaction.
135    pub input_state: ObjectIn,
136    /// State of the object in the store after this transaction.
137    pub output_state: ObjectOut,
138    /// Whether this object ID is created or deleted in this transaction.
139    /// This information isn't required by the protocol but is useful for
140    /// providing more detailed semantics on object changes.
141    pub id_operation: IdOperation,
142}
143
144impl From<iota_sdk::types::ChangedObject> for ChangedObject {
145    fn from(value: iota_sdk::types::ChangedObject) -> Self {
146        Self {
147            object_id: Arc::new(value.object_id.into()),
148            input_state: value.input_state.into(),
149            output_state: value.output_state.into(),
150            id_operation: value.id_operation,
151        }
152    }
153}
154
155impl From<ChangedObject> for iota_sdk::types::ChangedObject {
156    fn from(value: ChangedObject) -> Self {
157        Self {
158            object_id: **value.object_id,
159            input_state: value.input_state.into(),
160            output_state: value.output_state.into(),
161            id_operation: value.id_operation,
162        }
163    }
164}

199/// Type of unchanged shared object
200///
201/// # BCS
202///
203/// The BCS serialized form for this type is defined by the following ABNF:
204///
205/// ```text
206/// unchanged-shared-object-kind =  read-only-root
207///                              =/ mutate-deleted
208///                              =/ read-deleted
209///                              =/ cancelled
210///                              =/ per-epoch-config
211///
212/// read-only-root      = %x00 u64 digest
213/// mutate-deleted      = %x01 u64
214/// read-deleted        = %x02 u64
215/// cancelled           = %x03 u64
216/// per-epoch-config    = %x04
217/// ```
218#[derive(uniffi::Enum)]
219pub enum UnchangedSharedKind {
220    /// Read-only shared objects from the input. We don't really need
221    /// ObjectDigest for protocol correctness, but it will make it easier to
222    /// verify untrusted read.
223    ReadOnlyRoot { version: u64, digest: Arc<Digest> },
224    /// Deleted shared objects that appear mutably/owned in the input.
225    MutateDeleted { version: u64 },
226    /// Deleted shared objects that appear as read-only in the input.
227    ReadDeleted { version: u64 },
228    /// Shared objects in cancelled transaction. The sequence number embed
229    /// cancellation reason.
230    Cancelled { version: u64 },
231    /// Read of a per-epoch config object that should remain the same during an
232    /// epoch.
233    PerEpochConfig,
234}
235
236impl From<iota_sdk::types::UnchangedSharedKind> for UnchangedSharedKind {
237    fn from(value: iota_sdk::types::UnchangedSharedKind) -> Self {
238        match value {
239            iota_sdk::types::UnchangedSharedKind::ReadOnlyRoot { version, digest } => {
240                Self::ReadOnlyRoot {
241                    version,
242                    digest: Arc::new(digest.into()),
243                }
244            }
245            iota_sdk::types::UnchangedSharedKind::MutateDeleted { version } => {
246                Self::MutateDeleted { version }
247            }
248            iota_sdk::types::UnchangedSharedKind::ReadDeleted { version } => {
249                Self::ReadDeleted { version }
250            }
251            iota_sdk::types::UnchangedSharedKind::Cancelled { version } => {
252                Self::Cancelled { version }
253            }
254            iota_sdk::types::UnchangedSharedKind::PerEpochConfig => Self::PerEpochConfig,
255            _ => unimplemented!("a new enum variant was added and needs to be handled"),
256        }
257    }
258}
259
260impl From<UnchangedSharedKind> for iota_sdk::types::UnchangedSharedKind {
261    fn from(value: UnchangedSharedKind) -> Self {
262        match value {
263            UnchangedSharedKind::ReadOnlyRoot { version, digest } => Self::ReadOnlyRoot {
264                version,
265                digest: **digest,
266            },
267            UnchangedSharedKind::MutateDeleted { version } => Self::MutateDeleted { version },
268            UnchangedSharedKind::ReadDeleted { version } => Self::ReadDeleted { version },
269            UnchangedSharedKind::Cancelled { version } => Self::Cancelled { version },
270            UnchangedSharedKind::PerEpochConfig => Self::PerEpochConfig,
271        }
272    }
273}

275/// State of an object prior to execution
276///
277/// If an object exists (at root-level) in the store prior to this transaction,
278/// it should be Data, otherwise it's Missing, e.g. wrapped objects should be
279/// Missing.
280///
281/// # BCS
282///
283/// The BCS serialized form for this type is defined by the following ABNF:
284///
285/// ```text
286/// object-in = object-in-missing / object-in-data
287///
288/// object-in-missing = %x00
289/// object-in-data    = %x01 u64 digest owner
290/// ```
291#[derive(uniffi::Enum)]
292pub enum ObjectIn {
293    Missing,
294    /// The old version, digest and owner.
295    Data {
296        version: u64,
297        digest: Arc<Digest>,
298        owner: Arc<Owner>,
299    },
300}
301
302impl From<iota_sdk::types::ObjectIn> for ObjectIn {
303    fn from(value: iota_sdk::types::ObjectIn) -> Self {
304        match value {
305            iota_sdk::types::ObjectIn::Missing => Self::Missing,
306            iota_sdk::types::ObjectIn::Data {
307                version,
308                digest,
309                owner,
310            } => Self::Data {
311                version,
312                digest: Arc::new(digest.into()),
313                owner: Arc::new(owner.into()),
314            },
315            _ => unimplemented!("a new enum variant was added and needs to be handled"),
316        }
317    }
318}
319
320impl From<ObjectIn> for iota_sdk::types::ObjectIn {
321    fn from(value: ObjectIn) -> Self {
322        match value {
323            ObjectIn::Missing => Self::Missing,
324            ObjectIn::Data {
325                version,
326                digest,
327                owner,
328            } => Self::Data {
329                version,
330                digest: **digest,
331                owner: **owner,
332            },
333        }
334    }
335}

337/// State of an object after execution
338///
339/// # BCS
340///
341/// The BCS serialized form for this type is defined by the following ABNF:
342///
343/// ```text
344/// object-out  =  object-out-missing
345///             =/ object-out-object-write
346///             =/ object-out-package-write
347///
348///
349/// object-out-missing        = %x00
350/// object-out-object-write   = %x01 digest owner
351/// object-out-package-write  = %x02 version digest
352/// ```
353#[derive(uniffi::Enum)]
354pub enum ObjectOut {
355    /// Same definition as in ObjectIn.
356    Missing,
357    /// Any written object, including all of mutated, created, unwrapped today.
358    ObjectWrite {
359        digest: Arc<Digest>,
360        owner: Arc<Owner>,
361    },
362    /// Packages writes need to be tracked separately with version because
363    /// we don't use lamport version for package publish and upgrades.
364    PackageWrite { version: u64, digest: Arc<Digest> },
365}
366
367impl From<iota_sdk::types::ObjectOut> for ObjectOut {
368    fn from(value: iota_sdk::types::ObjectOut) -> Self {
369        match value {
370            iota_sdk::types::ObjectOut::Missing => Self::Missing,
371            iota_sdk::types::ObjectOut::ObjectWrite { digest, owner } => Self::ObjectWrite {
372                digest: Arc::new(digest.into()),
373                owner: Arc::new(owner.into()),
374            },
375            iota_sdk::types::ObjectOut::PackageWrite { version, digest } => Self::PackageWrite {
376                version,
377                digest: Arc::new(digest.into()),
378            },
379            _ => unimplemented!("a new enum variant was added and needs to be handled"),
380        }
381    }
382}
383
384impl From<ObjectOut> for iota_sdk::types::ObjectOut {
385    fn from(value: ObjectOut) -> Self {
386        match value {

396        }
397    }
398}

400/// Defines what happened to an ObjectId during execution
401///
402/// # BCS
403///
404/// The BCS serialized form for this type is defined by the following ABNF:
405///
406/// ```text
407/// id-operation =  id-operation-none
408///              =/ id-operation-created
409///              =/ id-operation-deleted
410///
411/// id-operation-none       = %x00
412/// id-operation-created    = %x01
413/// id-operation-deleted    = %x02
414/// ```
415#[uniffi::remote(Enum)]
416#[non_exhaustive]
417#[repr(u8)]
418pub enum IdOperation {
419    None,
420    Created,
421    Deleted,
422}
