6/// Unique identifier for an Account on the IOTA blockchain.
7///
8/// An `Address` is a 32-byte pseudonymous identifier used to uniquely identify
9/// an account and asset-ownership on the IOTA blockchain. Often, human-readable
10/// addresses are encoded in hexadecimal with a `0x` prefix. For example, this
11/// is a valid IOTA address:
12/// `0x02a212de6a9dfa3a69e22387acfbafbb1a9e591bd9d636e7895dcfc8de05f331`.
13///
14/// # Deriving an Address
15///
16/// Addresses are cryptographically derived from a number of user account
17/// authenticators, the simplest of which is an
18/// `Ed25519PublicKey`.
19///
20/// Deriving an address consists of the Blake2b256 hash of the sequence of bytes
21/// of its corresponding authenticator, prefixed with a domain-separator (except
22/// ed25519, for compatibility reasons). For each other authenticator, this
23/// domain-separator is the single byte-value of its
24/// `SignatureScheme` flag. E.g. `hash(signature schema flag || authenticator
25/// bytes)`.
26///
27/// Each authenticator has a method for deriving its `Address` as well as
28/// documentation for the specifics of how the derivation is done. See
29/// `Ed25519PublicKey::derive_address` for an example.
30///
31/// ## Relationship to ObjectIds
32///
33/// `ObjectId`s and `Address`es share the same 32-byte addressable space but
34/// are derived leveraging different domain-separator values to ensure that,
35/// cryptographically, there won't be any overlap, e.g. there can't be a
36/// valid `Object` who's `ObjectId` is equal to that of the `Address` of a user
37/// account.
38///
39/// # BCS
40///
41/// An `Address`'s BCS serialized form is defined by the following:
42///
43/// ```text
44/// address = 32OCTET
45/// ```
46#[derive(
47    Debug,
48    Hash,
49    PartialEq,
50    Eq,
51    PartialOrd,
52    Ord,
53    derive_more::Display,
54    derive_more::From,
55    derive_more::Deref,
56    uniffi::Object,
57)]
58#[uniffi::export(Debug, Display, Eq, Hash)]
59pub struct Address(pub iota_sdk::types::Address);
60
61#[uniffi::export]
62impl Address {
63    #[uniffi::constructor]
64    pub fn from_bytes(bytes: Vec<u8>) -> Result<Self> {
65        Ok(Self(iota_sdk::types::Address::from_bytes(bytes)?))
66    }
67
68    #[uniffi::constructor]
69    /// Parses an Address from a hex string, with or without a `0x` prefix.
70    /// The string can be of variable length; if it's shorter than 64 hex
71    /// characters, it will be left-padded with `0`s.
72    pub fn from_hex(hex: &str) -> Result<Self> {
73        Ok(Self(iota_sdk::types::Address::from_hex(hex)?))
74    }
75
76    #[uniffi::constructor]
77    pub fn generate() -> Self {
78        let mut rng = rand::thread_rng();
79        Self(iota_sdk::types::Address::generate(&mut rng))
80    }
81
82    pub fn to_bytes(&self) -> Vec<u8> {
83        self.0.as_bytes().to_vec()
84    }
85
86    pub fn to_hex(&self) -> String {
87        self.0.to_hex()
88    }
89
90    /// Returns the string representation of this address using the
91    /// canonical display, with or without a `0x` prefix.
92    pub fn to_canonical_string(&self, with_prefix: bool) -> String {
93        self.0.to_canonical_string(with_prefix)
94    }
95
96    /// Returns the shortest possible string representation of the address (i.e.
97    /// with leading zeroes trimmed).
98    pub fn to_short_string(&self, with_prefix: bool) -> String {
99        self.0.to_short_string(with_prefix)
100    }
101}

103macro_rules! named_address {
104    ($($constant:ident),+ $(,)?) => {
105        paste::paste! {
106            #[uniffi::export]
107            impl Address {$(
108                #[uniffi::constructor]
109                pub const fn [< $constant:lower >]() -> Self {
110                    Self(iota_sdk::types::Address::$constant)
111                }
112            )+}
113        }
114    }
115}
