13/// A zklogin authenticator
14///
15/// # BCS
16///
17/// The BCS serialized form for this type is defined by the following ABNF:
18///
19/// ```text
20/// zklogin-bcs = bytes             ; contents are defined by <zklogin-authenticator>
21/// zklogin     = zklogin-flag
22///               zklogin-inputs
23///               u64               ; max epoch
24///               simple-signature    
25/// ```
26///
27/// Note: Due to historical reasons, signatures are serialized slightly
28/// different from the majority of the types in IOTA. In particular if a
29/// signature is ever embedded in another structure it generally is serialized
30/// as `bytes` meaning it has a length prefix that defines the length of
31/// the completely serialized signature.
32#[derive(Debug, PartialEq, Eq, derive_more::From, uniffi::Object)]
33#[uniffi::export(Debug, Eq)]
34pub struct ZkLoginAuthenticator(pub iota_sdk::types::ZkLoginAuthenticator);
35
36#[uniffi::export]
37impl ZkLoginAuthenticator {
38    #[uniffi::constructor]
39    pub fn new(inputs: &ZkLoginInputs, max_epoch: u64, signature: &SimpleSignature) -> Self {
40        Self(iota_sdk::types::ZkLoginAuthenticator {
41            inputs: inputs.0.clone(),
42            max_epoch,
43            signature: signature.0.clone(),
44        })
45    }
46
47    pub fn inputs(&self) -> ZkLoginInputs {
48        self.0.inputs.clone().into()
49    }
50
51    pub fn max_epoch(&self) -> u64 {
52        self.0.max_epoch
53    }
54
55    pub fn signature(&self) -> SimpleSignature {
56        self.0.signature.clone().into()
57    }
58}

60/// Public Key equivalent for Zklogin authenticators
61///
62/// A `ZkLoginPublicIdentifier` is the equivalent of a public key for other
63/// account authenticators, and contains the information required to derive the
64/// onchain account `Address` for a Zklogin authenticator.
65///
66/// ## Note
67///
68/// Due to a historical bug that was introduced in the IOTA Typescript SDK when
69/// the zklogin authenticator was first introduced, there are now possibly two
70/// "valid" addresses for each zklogin authenticator depending on the
71/// bit-pattern of the `address_seed` value.
72///
73/// The original bug incorrectly derived a zklogin's address by stripping any
74/// leading zero-bytes that could have been present in the 32-byte length
75/// `address_seed` value prior to hashing, leading to a different derived
76/// address. This incorrectly derived address was presented to users of various
77/// wallets, leading them to sending funds to these addresses that they couldn't
78/// access. Instead of letting these users lose any assets that were sent to
79/// these addresses, the IOTA network decided to change the protocol to allow
80/// for a zklogin authenticator who's `address_seed` value had leading
81/// zero-bytes be authorized to sign for both the addresses derived from both
82/// the unpadded and padded `address_seed` value.
83///
84/// # BCS
85///
86/// The BCS serialized form for this type is defined by the following ABNF:
87///
88/// ```text
89/// zklogin-public-identifier-bcs = bytes ; where the contents are defined by
90///                                       ; <zklogin-public-identifier>
91///
92/// zklogin-public-identifier = zklogin-public-identifier-iss
93///                             address-seed
94///
95/// zklogin-public-identifier-unpadded = zklogin-public-identifier-iss
96///                                      address-seed-unpadded
97///
98/// ; The iss, or issuer, is a utf8 string that is less than 255 bytes long
99/// ; and is serialized with the iss's length in bytes as a u8 followed by
100/// ; the bytes of the iss
101/// zklogin-public-identifier-iss = u8 *255(OCTET)
102///
103/// ; A Bn254FieldElement serialized as a 32-byte big-endian value
104/// address-seed = 32(OCTET)
105///
106/// ; A Bn254FieldElement serialized as a 32-byte big-endian value
107/// ; with any leading zero bytes stripped
108/// address-seed-unpadded = %x00 / %x01-ff *31(OCTET)
109/// ```
110#[derive(Debug, PartialEq, Eq, derive_more::From, uniffi::Object)]
111#[uniffi::export(Debug, Eq)]
112pub struct ZkLoginPublicIdentifier(pub iota_sdk::types::ZkLoginPublicIdentifier);
113
114#[uniffi::export]
115impl ZkLoginPublicIdentifier {
116    #[uniffi::constructor]
117    pub fn new(iss: String, address_seed: &Bn254FieldElement) -> Result<Self> {
118        iota_sdk::types::ZkLoginPublicIdentifier::new(iss, address_seed.0.clone())
119            .ok_or_else(|| SdkFfiError::custom("iss length must be <= 255"))
120            .map(Self)
121    }
122
123    pub fn iss(&self) -> String {
124        self.0.iss().to_owned()
125    }
126
127    pub fn address_seed(&self) -> Bn254FieldElement {
128        self.0.address_seed().clone().into()
129    }
130
131    /// Derive an `Address` from this `ZkLoginPublicIdentifier` by hashing the
132    /// byte length of the `iss` followed by the `iss` bytes themselves and
133    /// the full 32 byte `address_seed` value, all prefixed with the zklogin
134    /// `SignatureScheme` flag (`0x05`).
135    ///
136    /// `hash( 0x05 || iss_bytes_len || iss_bytes || 32_byte_address_seed )`
137    pub fn derive_address_padded(&self) -> Address {
138        self.0.derive_address_padded().into()
139    }
140
141    /// Derive an `Address` from this `ZkLoginPublicIdentifier` by hashing the
142    /// byte length of the `iss` followed by the `iss` bytes themselves and
143    /// the `address_seed` bytes with any leading zero-bytes stripped, all
144    /// prefixed with the zklogin `SignatureScheme` flag (`0x05`).
145    ///
146    /// `hash( 0x05 || iss_bytes_len || iss_bytes ||
147    /// unpadded_32_byte_address_seed )`
148    pub fn derive_address_unpadded(&self) -> Address {
149        self.0.derive_address_unpadded().into()
150    }
151
152    /// Provides an iterator over the addresses that correspond to this zklogin
153    /// authenticator.
154    ///
155    /// In the majority of instances this will only yield a single address,
156    /// except for the instances where the `address_seed` value has a
157    /// leading zero-byte, in such cases the returned iterator will yield
158    /// two addresses.
159    pub fn derive_address(&self) -> Vec<Arc<Address>> {
160        self.0
161            .derive_address()
162            .map(Into::into)
163            .map(Arc::new)
164            .collect()
165    }
166}

350/// A point on the BN254 elliptic curve.
351///
352/// This is a 32-byte, or 256-bit, value that is generally represented as
353/// radix10 when a human-readable display format is needed, and is represented
354/// as a 32-byte big-endian value while in memory.
355///
356/// # BCS
357///
358/// The BCS serialized form for this type is defined by the following ABNF:
359///
360/// ```text
361/// bn254-field-element = *DIGIT ; which is then interpreted as a radix10 encoded 32-byte value
362/// ```
363#[derive(Debug, PartialEq, Eq, derive_more::From, uniffi::Object)]
364#[uniffi::export(Debug, Eq)]
365pub struct Bn254FieldElement(pub iota_sdk::types::Bn254FieldElement);
366
367#[uniffi::export]
368impl Bn254FieldElement {
369    #[uniffi::constructor]
370    pub fn from_bytes(bytes: Vec<u8>) -> Result<Self> {
371        Ok(Self(iota_sdk::types::Bn254FieldElement::new(
372            bytes.try_into().map_err(|v: Vec<u8>| {
373                SdkFfiError::custom(format!("expected bytes of length 32, found {}", v.len()))
374            })?,
375        )))
376    }
377
378    #[uniffi::constructor]
379    pub fn from_str(s: &str) -> Result<Self> {
380        Ok(Self(s.parse()?))
381    }
382
383    #[uniffi::constructor]
384    pub fn from_str_radix_10(s: &str) -> Result<Self> {
385        Ok(Self(iota_sdk::types::Bn254FieldElement::from_str_radix_10(
386            s,
387        )?))
388    }
389
390    pub fn unpadded(&self) -> Vec<u8> {
391        self.0.unpadded().to_vec()
392    }
393
394    pub fn padded(&self) -> Vec<u8> {
395        self.0.padded().to_vec()
396    }
397}

416/// A JSON Web Key
417///
418/// Struct that contains info for a JWK. A list of them for different kids can
419/// be retrieved from the JWK endpoint (e.g. <https://www.googleapis.com/oauth2/v3/certs>).
420/// The JWK is used to verify the JWT token.
421///
422/// # BCS
423///
424/// The BCS serialized form for this type is defined by the following ABNF:
425///
426/// ```text
427/// jwk = string string string string
428/// ```
429#[uniffi::remote(Record)]
430pub struct Jwk {
431    /// Key type parameter, <https://datatracker.ietf.org/doc/html/rfc7517#section-4.1>
432    pub kty: String,
433    /// RSA public exponent, <https://datatracker.ietf.org/doc/html/rfc7517#section-9.3>
434    pub e: String,
435    /// RSA modulus, <https://datatracker.ietf.org/doc/html/rfc7517#section-9.3>
436    pub n: String,
437    /// Algorithm parameter, <https://datatracker.ietf.org/doc/html/rfc7517#section-4.4>
438    pub alg: String,
439}
