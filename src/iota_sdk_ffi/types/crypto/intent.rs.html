8/// Intent errors.
9#[uniffi::remote(Enum)]
10#[uniffi::export(Debug)]
11#[non_exhaustive]
12pub enum IntentError {
13    /// Invalid bytes for Intent
14    Bytes,
15    /// Invalid hex string for Intent
16    Hex,
17    /// Invalid Scope for Intent
18    Scope,
19    /// Invalid Version for Intent
20    Version,
21    /// Invalid AppId for Intent
22    AppId,
23}

25/// Byte signifying the scope of an Intent
26///
27/// This enum specifies the intent scope. Two intents for different scopes
28/// should never collide, so no signature provided for one intent scope can be
29/// used for another, even when the serialized data itself may be the same.
30///
31/// # BCS
32///
33/// The BCS serialized form for this type is defined by the following ABNF:
34///
35/// ```text
36/// intent-scope = u8
37/// ```
38#[uniffi::remote(Enum)]
39#[uniffi::export(Debug, Eq, Hash)]
40#[non_exhaustive]
41pub enum IntentScope {
42    /// Used for a user signature on a transaction data.
43    TransactionData = 0,
44    /// Used for an authority signature on transaction effects.
45    TransactionEffects = 1,
46    /// Used for an authority signature on a checkpoint summary.
47    CheckpointSummary = 2,
48    /// Used for a user signature on a personal message.
49    PersonalMessage = 3,
50    /// Used for an authority signature on a user signed transaction.
51    SenderSignedTransaction = 4,
52    /// Used as a signature representing an authority's proof of possession of
53    /// its authority key.
54    ProofOfPossession = 5,
55    /// Deprecated. Should not be reused. Introduced for bridge purposes but was
56    /// never included in messages.
57    BridgeEventDeprecated = 6,
58    /// Used for consensus authority signature on block's digest.
59    ConsensusBlock = 7,
60    /// Used for reporting peer addresses in discovery
61    DiscoveryPeers = 8,
62    /// Used for authority capabilities from non-committee authorities.
63    AuthorityCapabilities = 9,
64}

66/// Byte signifying the version of an Intent
67///
68/// The version here is to distinguish between signing different versions of the
69/// struct or enum. Serialized output between two different versions of the same
70/// struct/enum might accidentally (or maliciously on purpose) match.
71///
72/// # BCS
73///
74/// The BCS serialized form for this type is defined by the following ABNF:
75///
76/// ```text
77/// intent-version = u8
78/// ```
79#[uniffi::remote(Enum)]
80#[uniffi::export(Debug, Eq, Hash)]
81#[non_exhaustive]
82pub enum IntentVersion {
83    V0 = 0,
84}
85
86/// Byte signifying the application id of an Intent
87///
88/// This enum specifies the application ID. Two intents in two different
89/// applications (i.e., IOTA, Ethereum etc) should never collide, so
90/// that even when a signing key is reused, nobody can take a signature
91/// designated for app_1 and present it as a valid signature for an (any) intent
92/// in app_2.
93///
94/// # BCS
95///
96/// The BCS serialized form for this type is defined by the following ABNF:
97///
98/// ```text
99/// intent-app-id = u8
100/// ```
101#[uniffi::remote(Enum)]
102#[uniffi::export(Debug, Eq, Hash)]
103#[non_exhaustive]
104pub enum IntentAppId {
105    Iota = 0,
106    Consensus = 1,
107}
108
109/// A Signing Intent
110///
111/// An intent is a compact struct that serves as the domain separator for a
112/// message that a signature commits to. It consists of three parts:
113///     1. IntentScope (what the type of the message is)
114///     2. IntentVersion
115///     3. IntentAppId (what application the signature refers to).
116///
117/// The serialization of an Intent is a 3-byte array where each field is
118/// represented by a byte and it is prepended onto a message before it is signed
119/// in IOTA.
120///
121/// # BCS
122///
123/// The BCS serialized form for this type is defined by the following ABNF:
124///
125/// ```text
126/// intent = intent-scope intent-version intent-app-id
127/// ```
128#[derive(Debug, PartialEq, Eq, Hash, uniffi::Object, derive_more::From)]
129#[uniffi::export(Debug, Eq, Hash)]
130pub struct Intent(pub iota_sdk::types::Intent);
131
132#[uniffi::export]
133impl Intent {
134    /// Create a new signing intent.
135    #[uniffi::constructor]
136    pub fn new(scope: IntentScope, version: IntentVersion, app_id: IntentAppId) -> Self {
137        Self(iota_sdk::types::Intent::new(scope, version, app_id))
138    }
139
140    /// Create a new IOTA app signing intent.
141    #[uniffi::constructor]
142    pub fn new_iota_app(scope: IntentScope) -> Self {
143        Self::new(scope, IntentVersion::V0, IntentAppId::Iota)
144    }
145
146    /// Create a new Consensus app signing intent.
147    #[uniffi::constructor]
148    pub fn new_consensus_app(scope: IntentScope) -> Self {
149        Self::new(scope, IntentVersion::V0, IntentAppId::Consensus)
150    }
151
152    /// Create a new IOTA transaction signing intent.
153    #[uniffi::constructor]
154    pub fn new_iota_transaction() -> Self {
155        Self::new(
156            IntentScope::TransactionData,
157            IntentVersion::V0,
158            IntentAppId::Iota,
159        )
160    }
161
162    /// Create a new IOTA personal message signing intent.
163    #[uniffi::constructor]
164    pub fn new_personal_message() -> Self {
165        Self::new(
166            IntentScope::PersonalMessage,
167            IntentVersion::V0,
168            IntentAppId::Iota,
169        )
170    }
171
172    /// Create a signing intent from bytes.
173    #[uniffi::constructor]
174    pub fn from_bytes(bytes: &[u8]) -> Result<Intent, IntentError> {
175        Ok(Intent(iota_sdk::types::crypto::Intent::from_bytes(bytes)?))
176    }
177
178    /// Create a signing intent from a hex string.
179    #[uniffi::constructor]
180    pub fn from_hex(hex: &str) -> Result<Intent, IntentError> {
181        Ok(Intent(iota_sdk::types::crypto::Intent::from_str(hex)?))
182    }
183
184    /// Get the scope of the signing intent.
185    pub fn scope(&self) -> IntentScope {
186        self.0.scope
187    }
188
189    /// Get the version of the signing intent.
190    pub fn version(&self) -> IntentVersion {
191        self.0.version
192    }
193
194    /// Get the app id of the signing intent.
195    pub fn app_id(&self) -> IntentAppId {
196        self.0.app_id
197    }
198
199    /// Convert the signing intent to bytes.
200    pub fn to_bytes(&self) -> Vec<u8> {
201        self.0.to_bytes().to_vec()
202    }
203}

205/// A 1-byte domain separator for hashing Object ID in IOTA. It starts from
206/// 0xf0 to ensure no hashing collision for any ObjectID vs IotaAddress which is
207/// derived as the hash of `flag || pubkey`.
208#[uniffi::remote(Enum)]
209#[uniffi::export(Eq, Debug, Hash)]
210#[non_exhaustive]
211pub enum HashingIntentScope {
212    ChildObjectId = 0xf0,
213    RegularObjectId = 0xf1,
214}

216/// A personal message that wraps around a byte array.
217#[derive(Debug, PartialEq, Eq, derive_more::From, uniffi::Object)]
218#[uniffi::export(Debug, Eq)]
219pub struct PersonalMessage(pub(crate) iota_sdk::types::PersonalMessage<'static>);
220
221#[uniffi::export]
222impl PersonalMessage {
223    /// Create a new personal message from bytes.
224    #[uniffi::constructor]
225    pub fn new(message_bytes: &[u8]) -> Self {
226        Self(iota_sdk::types::PersonalMessage(std::borrow::Cow::Owned(
227            message_bytes.to_vec(),
228        )))
229    }
230
231    /// Get the message as bytes.
232    pub fn message_bytes(&self) -> Vec<u8> {
233        self.0.0.clone().into_owned()
234    }
235
236    /// Get the signing digest as bytes.
237    pub fn signing_digest(&self) -> Vec<u8> {
238        self.0.signing_digest().to_vec()
239    }
240
241    /// Get the signing digest as hex string.
242    pub fn signing_digest_hex(&self) -> String {
243        self.0.signing_digest_hex()
244    }
