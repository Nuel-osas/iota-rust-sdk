4use std::sync::Arc;
5
6use crate::types::{
7    address::Address,
8    object::{ObjectId, ObjectReference},
9    transaction::Input,
10    type_tag::TypeTag,
11};
12
13/// MoveAuthenticator is a signature variant that enables a method of
14/// authentication through Move code. This type represents the data received
15/// by the Move authenticate function during the Account Abstraction
16/// authentication flow.
17#[derive(Debug, derive_more::From, uniffi::Object)]
18pub struct MoveAuthenticator(pub iota_sdk::types::MoveAuthenticator);
19
20#[uniffi::export]
21impl MoveAuthenticator {
22    /// Create a new move authenticator from an immutable object.
23    #[uniffi::constructor]
24    pub fn new_immutable(
25        call_args: Vec<Arc<Input>>,
26        type_args: Vec<Arc<TypeTag>>,
27        object_to_authenticate: ObjectReference,
28    ) -> Self {
29        Self(iota_sdk::types::MoveAuthenticator::new_immutable(
30            call_args.into_iter().map(|v| v.0.clone()).collect(),
31            type_args.into_iter().map(|v| v.0.clone()).collect(),
32            object_to_authenticate.into(),
33        ))
34    }
35
36    /// Create a new move authenticator from a shared object.
37    #[uniffi::constructor]
38    pub fn new_shared(
39        call_args: Vec<Arc<Input>>,
40        type_args: Vec<Arc<TypeTag>>,
41        object_to_authenticate: &ObjectId,
42        initial_shared_version: u64,
43    ) -> Self {
44        Self(iota_sdk::types::MoveAuthenticator::new_shared(
45            call_args.into_iter().map(|v| v.0.clone()).collect(),
46            type_args.into_iter().map(|v| v.0.clone()).collect(),
47            **object_to_authenticate,
48            initial_shared_version,
49        ))
50    }
51
52    pub fn address(&self) -> Address {
53        self.0.address().into()
54    }
55
56    pub fn call_args(&self) -> Vec<Arc<Input>> {
57        self.0
58            .call_args()
59            .iter()
60            .cloned()
61            .map(Into::into)
62            .map(Arc::new)
63            .collect()
64    }
65
66    pub fn type_args(&self) -> Vec<Arc<TypeTag>> {
67        self.0
68            .type_args()
69            .iter()
70            .cloned()
71            .map(Into::into)
72            .map(Arc::new)
73            .collect()
74    }
75
76    pub fn object_to_authenticate(&self) -> Input {
77        self.0.object_to_authenticate().clone().into()
78    }
79}
